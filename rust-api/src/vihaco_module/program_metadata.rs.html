4//! Standard program-header schema and section-local metadata validation.
5//!
6//! Every table is supplied by the header, even when empty. Array order defines
7//! table indices; names and files are explicit string-table indices. Loading
8//! never interns strings, invents signatures, or infers an entrypoint.
9//!
10//! The header says *what* each function is; the section's symbol table says
11//! *where* it is. A function declaration is bound to the `@name:` symbol with
12//! the same name, and its executable range runs from that symbol to the next
13//! function symbol (or the end of the code).
14//!
15//! Constants use their type's ordinary `Parse` implementation. Standard `Value`
16//! constants use typed SST literals such as `u64 42` or `string 0`; `f32_bits`
17//! and `f64_bits` followed by unsigned integers preserve all IEEE-754 encodings.
18//! Custom extra records use the same delimited metadata derive as the header.
19
20use std::collections::{BTreeMap, HashMap};

22use crate::module::{FunctionDecl, FunctionInfo, LocalModule, NoInfo, SourceSymbolInfo, Symbol};
23use vihaco_abi::{Type, Value};

25/// All non-code fields of a `LocalModule` that a section header supplies.
26/// The ordinary metadata derive owns object parsing, including nested records.
27#[derive(Debug, Clone, PartialEq, vihaco_parser_derive::Parse)]
28#[syntax_class(metadata, head = "metadata", open = "{", close = "}")]
29pub struct ProgramMetadata<V = Value, Ty = Type, Info = NoInfo> {
30    pub functions: Vec<FunctionDecl<Ty>>,
31    pub constants: Vec<V>,
32    pub strings: Vec<String>,
33    pub main_function: Option<u32>,
34    pub source_symbols: Vec<SourceSymbolInfo>,
35    pub extra: Info,
36}
37
38fn check_index(index: u32, len: usize, what: &str) -> eyre::Result<()> {
39    eyre::ensure!(
40        (index as usize) < len,
41        "{what}: index {index} is outside table of length {len}"
42    );
43    Ok(())
44}
45
46/// Check a symbol table against its code length and index it by name.
47///
48/// Names must be unique within a section, and every address must lie within
49/// the code (an address equal to the code length marks the end).
50pub fn index_symbols(
51    symbols: &[Symbol],
52    instruction_count: usize,
53) -> eyre::Result<HashMap<&str, u32>> {
54    let mut table = HashMap::with_capacity(symbols.len());
55    for symbol in symbols {
56        eyre::ensure!(
57            symbol.address as usize <= instruction_count,
58            "symbol `@{}` at {} is outside the instruction stream of length {instruction_count}",
59            symbol.name,
60            symbol.address
61        );
62        eyre::ensure!(
63            table.insert(symbol.name.as_str(), symbol.address).is_none(),
64            "duplicate symbol `@{}`",
65            symbol.name
66        );
67    }
68    Ok(table)
69}

71impl<V, Ty, Info> ProgramMetadata<V, Ty, Info> {
72    /// Transform consumer-specific extra metadata while preserving every table.
73    pub fn map_extra<NewInfo>(
74        self,
75        map: impl FnOnce(Info) -> NewInfo,
76    ) -> ProgramMetadata<V, Ty, NewInfo> {
77        ProgramMetadata {
78            functions: self.functions,
79            constants: self.constants,
80            strings: self.strings,
81            main_function: self.main_function,
82            source_symbols: self.source_symbols,
83            extra: map(self.extra),
84        }
85    }
86
87    /// The name of function declaration `index`, resolved through `strings`.
88    pub fn function_name(&self, index: usize) -> eyre::Result<&str> {
89        let function = self
90            .functions
91            .get(index)
92            .ok_or_else(|| eyre::eyre!("metadata.functions[{index}] does not exist"))?;
93        self.strings
94            .get(function.name as usize)
95            .map(String::as_str)
96            .ok_or_else(|| {
97                eyre::eyre!(
98                    "metadata.functions[{index}].name: index {} is outside table of length {}",
99                    function.name,
100                    self.strings.len()
101                )
102            })
103    }
104
105    /// Validate table references and bind declarations to the symbol table,
106    /// without altering the supplied metadata.
107    pub fn validate(&self, symbols: &[Symbol], instruction_count: usize) -> eyre::Result<()>
108    where
109        Ty: Clone,
110    {
111        self.resolve_functions(symbols, instruction_count)
112            .map(|_| ())
113    }
114
115    /// Bind every function declaration to its symbol and derive its range.
116    ///
117    /// Returns runtime function records in metadata table order. A function
118    /// starts at its symbol's address and ends at the next function symbol's
119    /// address, or at `instruction_count` for the last function.
120    pub fn resolve_functions(
121        &self,
122        symbols: &[Symbol],
123        instruction_count: usize,
124    ) -> eyre::Result<Vec<FunctionInfo<Ty>>>
125    where
126        Ty: Clone,
127    {
128        eyre::ensure!(
129            u32::try_from(instruction_count).is_ok(),
130            "instruction stream exceeds u32 address space"
131        );
132        let table = index_symbols(symbols, instruction_count)?;
133        if let Some(main) = self.main_function {
134            check_index(main, self.functions.len(), "metadata.main_function")?;
135        }
136
137        let mut starts = Vec::with_capacity(self.functions.len());
138        let mut names = BTreeMap::new();
139        for (index, function) in self.functions.iter().enumerate() {
140            let path = format!("metadata.functions[{index}]");
141            let name = self.function_name(index)?;
142            check_index(function.file, self.strings.len(), &format!("{path}.file"))?;
143            if let Some(previous) = names.insert(name, index) {
144                eyre::bail!(
145                    "function `@{name}` is declared by both metadata.functions[{previous}] and {path}"
146                );
147            }
148            eyre::ensure!(
149                function.local_count as usize >= function.signature.params.len(),
150                "{path}.local_count: fewer slots than parameters"
151            );
152            for parameter in &function.signature.params {
153                check_index(
154                    parameter.name,
155                    self.strings.len(),
156                    &format!("{path}.signature.params.name"),
157                )?;
158            }
159            let start = *table.get(name).ok_or_else(|| {
160                eyre::eyre!("function `@{name}` ({path}) has no `@{name}:` symbol definition")
161            })?;
162            starts.push(start);
163        }
164
165        let mut boundaries = starts.clone();
166        boundaries.sort_unstable();
167        boundaries.dedup();
168        let end_of = |start: u32| {
169            boundaries
170                .iter()
171                .copied()
172                .find(|&boundary| boundary > start)
173                .unwrap_or(instruction_count as u32)
174        };
175
176        let mut seen = std::collections::BTreeSet::new();
177        for symbol in &self.source_symbols {
178            eyre::ensure!(
179                seen.insert(symbol.index),
180                "metadata.source_symbols: duplicate index {}",
181                symbol.index
182            );
183        }
184
185        Ok(self
186            .functions
187            .iter()
188            .zip(starts)
189            .map(|(function, start)| FunctionInfo {
190                name: function.name,
191                signature: function.signature.clone(),
192                local_count: function.local_count,
193                start_address: start,
194                end_address: end_of(start),
195                file: function.file,
196            })
197            .collect())
198    }
199}

201impl<Ty: Clone, Info> ProgramMetadata<Value, Ty, Info> {
202    /// Assemble the standard value model, checking the supplied tables once.
203    /// Custom value models can assemble `LocalModule` directly after
204    /// [`resolve_functions`](Self::resolve_functions).
205    pub fn into_module<I>(
206        self,
207        code: Vec<I>,
208        symbols: Vec<Symbol>,
209    ) -> eyre::Result<LocalModule<I, Value, Ty, Info>> {
210        let functions = self.resolve_functions(&symbols, code.len())?;
211        for (index, constant) in self.constants.iter().enumerate() {
212            match constant {
213                Value::String(id) => check_index(
214                    *id,
215                    self.strings.len(),
216                    &format!("metadata.constants[{index}] string"),
217                )?,
218                Value::FunctionRef(id) => check_index(
219                    *id,
220                    self.functions.len(),
221                    &format!("metadata.constants[{index}] function"),
222                )?,
223                _ => {}
224            }
225        }
226        Ok(LocalModule {
227            code,
228            functions,
229            symbols,
230            constants: self.constants,
231            strings: self.strings,
232            main_function: self.main_function,
233            source_symbols: self.source_symbols,
234            extra: self.extra,
235        })
236    }
