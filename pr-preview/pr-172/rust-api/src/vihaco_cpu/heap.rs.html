4use crate::Word;
5
6/// Append-only storage with a fixed capacity limit and a current length.
7/// `heap_alloc` creates a full object, and `heap_reserve` creates an empty one.
8/// Appends never overwrite elements or grow the allocation, and reads are
9/// bounded by `values.len()` rather than capacity.
10#[derive(Debug, Default)]
11struct HeapObject {
12    values: Vec<Word>,
13    capacity: usize,
14}
15
16impl Clone for HeapObject {
17    fn clone(&self) -> Self {
18        // Vec::clone may discard spare capacity. Preserve it so appending to a
19        // cloned CPU's heap also needs no allocation.
20        let mut values = Vec::with_capacity(self.capacity);
21        values.extend_from_slice(&self.values);
22        Self {
23            values,
24            capacity: self.capacity,
25        }
26    }
27}
28
29#[derive(Debug, Clone)]
30enum HeapSlot {
31    Alive(HeapObject),
32    Deallocated,
33}
34
35#[derive(Debug, Clone, Default)]
36pub(crate) struct Heap {
37    slots: Vec<HeapSlot>,
38    free_list: Vec<u32>,
39}
40
41impl Heap {
42    /// Allocate a full object; spare Vec capacity does not permit appending.
43    pub(crate) fn alloc(&mut self, values: impl Into<Vec<Word>>) -> u32 {
44        let values = values.into();
45        let capacity = values.len();
46        self.insert(HeapObject { values, capacity })
47    }
48
49    /// Allocate an empty object with an exact, fixed element limit.
50    /// Returns an error if its storage cannot be allocated.
51    pub(crate) fn reserve(&mut self, capacity: usize) -> eyre::Result<u32> {
52        let mut values = Vec::new();
53        values.try_reserve_exact(capacity)?;
54        Ok(self.insert(HeapObject { values, capacity }))
55    }
56
57    /// Append without reallocating or overwriting existing elements.
58    /// Returns an error for invalid, deallocated, or full objects.
59    pub(crate) fn push(&mut self, id: u32, value: Word) -> eyre::Result<()> {
60        let object = match self.slots.get_mut(id as usize) {
61            Some(HeapSlot::Alive(object)) => object,
62            Some(HeapSlot::Deallocated) => {
63                return Err(eyre::eyre!("heap object {} has been deallocated", id));
64            }
65            None => return Err(eyre::eyre!("invalid heap object id {}", id)),
66        };
67        eyre::ensure!(
68            object.values.len() < object.capacity,
69            "heap object {} is full (capacity {})",
70            id,
71            object.capacity
72        );
73        object.values.push(value);
74        Ok(())
75    }
76
77    fn insert(&mut self, object: HeapObject) -> u32 {
78        if let Some(id) = self.free_list.pop() {
79            self.slots[id as usize] = HeapSlot::Alive(object);
80            id
81        } else {
82            let id = self.slots.len() as u32;
83            self.slots.push(HeapSlot::Alive(object));
84            id
85        }
86    }
87
88    pub(crate) fn dealloc(&mut self, id: u32) -> eyre::Result<()> {
89        match self.slots.get_mut(id as usize) {
90            Some(slot @ HeapSlot::Alive(_)) => {
91                *slot = HeapSlot::Deallocated;
92                self.free_list.push(id);
93                Ok(())
94            }
95            Some(HeapSlot::Deallocated) => Err(eyre::eyre!(
96                "double-free: heap object {} already deallocated",
97                id
98            )),
99            None => Err(eyre::eyre!("invalid heap object id {}", id)),
100        }
101    }
102
103    pub(crate) fn get(&self, id: u32) -> eyre::Result<&[Word]> {
104        match self.slots.get(id as usize) {
105            Some(HeapSlot::Alive(object)) => Ok(&object.values),
106            Some(HeapSlot::Deallocated) => {
107                Err(eyre::eyre!("heap object {} has been deallocated", id))
108            }
109            None => Err(eyre::eyre!("invalid heap object id {}", id)),
110        }
111    }
112
113    pub(crate) fn clear(&mut self) {
114        self.slots.clear();
115        self.free_list.clear();
116    }
117
118    #[cfg(all(test, not(feature = "program-image-execution")))]
119    pub fn is_empty(&self) -> bool {
120        self.slots.is_empty()
121    }
122}

124#[cfg(test)]
125mod heap_tests {
126    use super::Heap;
127
128    #[test]
129    fn allocated_objects_are_full_even_with_spare_vec_capacity() {
130        let mut heap = Heap::default();
131        let mut values = Vec::with_capacity(8);
132        values.push(1);
133        let id = heap.alloc(values);
134        assert!(heap.push(id, 2).unwrap_err().to_string().contains("full"));
135        assert_eq!(heap.get(id).unwrap(), &[1]);
136    }
137
138    #[test]
139    fn cloned_heap_preserves_append_capacity_without_reallocation() {
140        let mut heap = Heap::default();
141        let id = heap.reserve(3).unwrap();
142        heap.push(id, 1).unwrap();
143        let mut cloned = heap.clone();
144        for heap in [&mut heap, &mut cloned] {
145            let storage = heap.get(id).unwrap().as_ptr();
146            heap.push(id, 2).unwrap();
147            heap.push(id, 3).unwrap();
148            assert_eq!(heap.get(id).unwrap().as_ptr(), storage);
149            assert_eq!(heap.get(id).unwrap(), &[1, 2, 3]);
150            assert!(heap.push(id, 4).is_err());
151        }
152    }
