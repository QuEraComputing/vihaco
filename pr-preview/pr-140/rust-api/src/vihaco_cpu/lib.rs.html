4//! Checked word-stack CPU component.
5//!
6//! Enable `vihaco-cpu/typed-declarations` to use the `cpu` runtime dialect and
7//! per-instruction `Execute` implementations. This forwards the framework feature;
8//! enabling the framework feature alone does not select CPU's typed declaration.
9//! Feature-off use retains the legacy instruction enums and `CPUMessage`.
10//!
11//! Typed instruction syntax retains readable literals until explicit lowering
12//! encodes them into runtime `Word` payloads. Branch and call targets accept
13//! numeric addresses or `@name` symbols; function references accept numeric
14//! indices or function symbols. String constants accept indices or quoted text.
15//! Loaders supply `lower::LabelLookup`, `lower::FunctionLookup`, and
16//! `lower::StringIntern` to resolve them. [`vihaco::ProgramImage`] implements
17//! string interning using its section string table. Heap references remain numeric.
18//! `label @name` is retained in the parsed function's label metadata without an
19//! executable slot. Labels, `function_start`, and `function_end` have no runtime
20//! instruction leaves.
21//! Composite source uses `device::dialect::mnemonic`, for example
22//! `host::cpu::const_i64 -42`. Generated aggregate bytecode codecs are separate work.
23//!
24//! ```
25//! # #[cfg(feature = "typed-declarations")]
26//! # {
27//! use vihaco::{Execute, Lower, Parse, Syntax, chumsky::Parser as _, traits::StackMemory};
28//! use vihaco_cpu::{CPU, cpu, encode_i64, StepOutcome};
29//! let instruction = cpu::ConstI64::<Syntax>::parser().parse("const_i64 -42")
30//!     .into_result().unwrap();
31//! let instruction = instruction.lower(&mut ()).unwrap();
32//! assert_eq!(instruction, cpu::ConstI64(encode_i64(-42)));
33//! let mut cpu = CPU::default();
34//! assert_eq!(cpu.execute(&instruction, ()).unwrap(), StepOutcome::Continue);
35//! assert_eq!(cpu.stack(), &[encode_i64(-42)]);
36//! # }
37//! ```
38//!
39//! Equal string contents reuse the same section-local index:
40//!
41//! ```
42//! # #[cfg(feature = "typed-declarations")]
43//! # {
44//! use vihaco::{Lower, NoContext, Parse, ProgramImage, Syntax, chumsky::Parser as _};
45//! use vihaco_cpu::cpu;
46//! let mut image = ProgramImage::<(), NoContext>::default();
47//! for _ in 0..2 {
48//!     let source = cpu::ConstString::<Syntax>::parser().parse("const_string \"hello\"")
49//!         .into_result().unwrap();
50//!     assert_eq!(source.lower(&mut image).unwrap(), cpu::ConstString(0));
51//! }
52//! assert_eq!(image.module.strings, ["hello"]);
53//! # }
54//! ```
