4//! Instruction-owned program raising: the inverse of [`lower`](super::lower).
5//!
6//! Each leaf whose source operands differ from its runtime operands raises
7//! with the capabilities it needs from the context, as
8//! [`vihaco_module::raising`] traits: branch targets name the symbol at their
9//! address, calls and function references name the function, string
10//! constants spell their text, and literal words are decoded back to the
11//! literal they were lowered from. A word that no literal lowers to is an
12//! error rather than a different value.
13
14use super::{
15    literal::{F32Literal, F64Literal, I32Literal, I64Literal},
16    operand::{StringOperand, Target},
17    program,
18};
19use vihaco_abi::{Raise, Syntax, word};
20use vihaco_module::raising::{FunctionNames, LabelNames, StringTable, decode_canonical};
21use vihaco_parser::{Ident, QuotedString};
22
23fn symbol(name: &str) -> Target {
24    Target::Symbol(Ident(name.to_owned()))
25}
26
27fn label<C: LabelNames + ?Sized>(address: u32, context: &C) -> eyre::Result<Target> {
28    context.label_name(address).map(symbol)
29}
30
31impl<C: LabelNames + ?Sized> Raise<C> for program::Branch {
32    type Syntax = program::Branch<Syntax>;
33    fn raise(&self, context: &C) -> eyre::Result<Self::Syntax> {
34        Ok(program::Branch(label(self.0, context)?))
35    }
36}
37
38impl<C: LabelNames + ?Sized> Raise<C> for program::ConditionalBranch {
39    type Syntax = program::ConditionalBranch<Syntax>;
40    fn raise(&self, context: &C) -> eyre::Result<Self::Syntax> {
41        // Look both targets up before failing, so that every address without
42        // a symbol is reported to the context.
43        let (then, otherwise) = (label(self.0, context), label(self.1, context));
44        Ok(program::ConditionalBranch(then?, otherwise?))
45    }
46}
47
48impl<C: FunctionNames + ?Sized> Raise<C> for program::Call {
49    type Syntax = program::Call<Syntax>;
50    fn raise(&self, context: &C) -> eyre::Result<Self::Syntax> {
51        Ok(program::Call(self.0, symbol(context.function_at(self.1)?)))
52    }
53}
54
55impl<C: FunctionNames + ?Sized> Raise<C> for program::ConstFunctionRef {
56    type Syntax = program::ConstFunctionRef<Syntax>;
57    fn raise(&self, context: &C) -> eyre::Result<Self::Syntax> {
58        let index = decode_canonical(
59            self.0,
60            word::decode_function_ref,
61            word::encode_function_ref,
62            "function reference",
63        )?;
64        Ok(program::ConstFunctionRef(symbol(
65            context.function_name(index)?,
66        )))
67    }
68}
69
70impl<C: StringTable + ?Sized> Raise<C> for program::ConstString {
71    type Syntax = program::ConstString<Syntax>;
72    fn raise(&self, context: &C) -> eyre::Result<Self::Syntax> {
73        let index = decode_canonical(
74            self.0,
75            word::decode_string_id,
76            word::encode_string_id,
77            "string",
78        )?;
79        let text = context.string(index)?.to_owned();
80        Ok(program::ConstString(StringOperand::Literal(QuotedString(
81            text,
82        ))))
83    }
84}
85
86// Literal decoding is instruction-owned raising and needs no context.
87macro_rules! raise_literal {
88    ($($instruction:ident => $decode:path, $encode:path, $what:literal, $wrap:expr;)*) => {$(
89        impl<C: ?Sized> Raise<C> for program::$instruction {
90            type Syntax = program::$instruction<Syntax>;
91            fn raise(&self, _: &C) -> eyre::Result<Self::Syntax> {
92                let value = decode_canonical(self.0, $decode, $encode, $what)?;
93                Ok(program::$instruction(($wrap)(value)))
94            }
95        }
96    )*};
97}
98
99raise_literal! {
100    ConstI32 => word::decode_i32, word::encode_i32, "i32", I32Literal;
101    ConstI64 => word::decode_i64, word::encode_i64, "i64", I64Literal;
102    ConstF32 => word::decode_f32, word::encode_f32, "f32", F32Literal;
103    ConstF64 => word::decode_f64, word::encode_f64, "f64", F64Literal;
104    ConstU32 => word::decode_u32, word::encode_u32, "u32", std::convert::identity;
105    ConstBool => word::decode_bool, word::encode_bool, "bool", std::convert::identity;
106    ConstHeapRef => word::decode_heap_ref, word::encode_heap_ref, "heap reference",
107        std::convert::identity;
108}

121    fn metadata() -> ProgramMetadata {
122        let function = |name| FunctionDecl {
123            name,
124            signature: Signature {
125                params: vec![],
126                ret: vec![],
127            },
128            local_count: 0,
129            file: 0,
130        };
131        ProgramMetadata {
132            strings: vec!["helper".into(), "main".into(), "hello".into()],
133            functions: vec![function(0), function(1)],
134            constants: vec![],
135            main_function: Some(1),
136            source_symbols: vec![],
137            extra: NoInfo,
138        }
139    }

141    fn symbols() -> Vec<Symbol> {
142        [("main", 0), ("done", 4), ("helper", 5)]
143            .map(|(name, address)| Symbol {
144                name: name.into(),
145                address,
146            })
147            .to_vec()
148    }

174    /// Raise `instruction`, check its source spelling, and lower it back.
175    fn round_trip<I>(instruction: I, spelled: &str) -> eyre::Result<()>
176    where
177        I: Raise<RaiseContext> + PartialEq + std::fmt::Debug,
178        I::Syntax: Lower<MetadataLoweringContext, Runtime = I> + Print,
179    {
180        let raised = instruction.raise(&RaiseContext::new(&module())?)?;
181        assert_eq!(raised.to_source().unwrap(), spelled);
182        let metadata = metadata();
183        let mut context = MetadataLoweringContext::new(&metadata, &symbols())?;
184        assert_eq!(raised.lower(&mut context)?, instruction);
185        Ok(())
186    }
187
188    #[test]
189    fn raising_names_what_lowering_resolved() -> eyre::Result<()> {
190        round_trip(program::Branch(4), "br @done")?;
191        round_trip(program::ConditionalBranch(0, 4), "cond_br @main, @done")?;
192        round_trip(program::Call(2, 5), "call 2, @helper")?;
193        round_trip(
194            program::ConstFunctionRef(word::encode_function_ref(1)),
195            "const_fn_ref @main",
196        )?;
197        round_trip(
198            program::ConstString(word::encode_string_id(2)),
199            "const_string \"hello\"",
200        )?;
201        round_trip(program::ConstI32(word::encode_i32(-7)), "const_i32 -7")?;
202        round_trip(program::ConstI64(word::encode_i64(-7)), "const_i64 -7")?;
203        round_trip(program::ConstF32(word::encode_f32(0.5)), "const_f32 0.5")?;
204        round_trip(program::ConstF64(word::encode_f64(-2.5)), "const_f64 -2.5")?;
205        round_trip(program::ConstU32(word::encode_u32(9)), "const_u32 9")?;
206        round_trip(
207            program::ConstBool(word::encode_bool(true)),
208            "const_bool true",
209        )?;
210        round_trip(
211            program::ConstHeapRef(word::encode_heap_ref(3)),
212            "const_heap_ref 3",
213        )
214    }

216    #[test]
217    fn raising_reports_what_has_no_name_or_spelling() -> eyre::Result<()> {
218        let module = module();
219        let context = RaiseContext::new(&module)?;
220        // Both missing branch targets are recorded for label synthesis.
221        assert!(program::ConditionalBranch(1, 2).raise(&context).is_err());
222        assert_eq!(context.unlabeled(), [1, 2]);
223        let error = program::Call(0, 4).raise(&context).unwrap_err();
224        assert_eq!(error.to_string(), "no function starts at address 0x0004");
225        assert!(
226            program::ConstString(word::encode_string_id(3))
227                .raise(&context)
228                .is_err()
229        );
230        assert!(
231            program::ConstFunctionRef(word::encode_function_ref(2))
232                .raise(&context)
233                .is_err()
234        );
235        let error = program::ConstBool(5).raise(&context).unwrap_err();
236        assert_eq!(
237            error.to_string(),
238            "word 0x5 is not the encoding of any bool literal"
239        );
240        assert!(program::ConstI32(u64::MAX).raise(&context).is_err());
241        assert!(program::ConstString(u64::MAX).raise(&context).is_err());
242        Ok(())
243    }
