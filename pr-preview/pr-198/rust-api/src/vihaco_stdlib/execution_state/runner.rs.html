11/// Enter main and execute through the composite's authored Steps.
12///
13/// Devices remain in place. Instructions borrow the caller-owned module; there
14/// is no instruction clone, generated Step, or imposed effect-application policy.
15///
16/// # Errors
17/// Returns entry, checked-fetch, or authored Dispatch/Step errors. Operand and
18/// component mutations made before an error are retained.
19pub fn execute<M, I, V, Ty, Info, S>(
20    module: &LocalModule<I, V, Ty, Info>,
21    machine: &mut M,
22    arguments: &[Word],
23) -> eyre::Result<()>
24where
25    M: HasProgram<Program = ExecutionState<S>>
26        + Dispatch<LocalModule<I, V, Ty, Info>, Instruction = I>,
27{
28    machine.program_mut().start(module, arguments)?;
29    resume(module, machine)
30}

32/// Continue an invocation until Halt/root return or an error.
33///
34/// Fetch establishes the sequential continuation before the Step runs. Program
35/// instructions can replace that continuation. A fault or terminal instruction
36/// leaves its original position available through [`ExecutionState::pc()`].
37///
38/// The caller must supply the same unchanged module used to start the active
39/// invocation. This checked API validates accesses, but does not enforce module
40/// identity. After an error, earlier operand consumption and effects remain;
41/// the caller is responsible for deciding whether the invocation can resume.
42///
43/// # Errors
44/// Rejects reentrant execution, invalid running state, invalid fetches, PC
45/// overflow, or a failure returned by the authored Dispatch/Step.
46pub fn resume<M, I, V, Ty, Info, S>(
47    module: &LocalModule<I, V, Ty, Info>,
48    machine: &mut M,
49) -> eyre::Result<()>
50where
51    M: HasProgram<Program = ExecutionState<S>>
52        + Dispatch<LocalModule<I, V, Ty, Info>, Instruction = I>,
53{
54    run(
55        machine,
56        |_| Ok(()),
57        |machine, ()| {
58            execute_instruction(module, machine, None)?;
59            Ok(ControlFlow::Continue(()))
60        },
61    )
62}

64pub(crate) fn run<M, S, State>(
65    machine: &mut M,
66    initialize: impl FnOnce(&mut M) -> eyre::Result<State>,
67    mut step: impl FnMut(&mut M, &mut State) -> eyre::Result<ControlFlow<()>>,
68) -> eyre::Result<()>
69where
70    M: HasProgram<Program = ExecutionState<S>>,
71{
72    let program = machine.program_mut();
73    eyre::ensure!(
74        !program.executing,
75        "program execution is already in progress"
76    );
77    program.executing = true;
78    let result = (|| {
79        let mut state = initialize(machine)?;
80        while matches!(machine.program().run_state, RunState::Runnable) {
81            if step(machine, &mut state)?.is_break() {
82                break;
83            }
84        }
85        Ok(())
86    })();
87    machine.program_mut().executing = false;
88    result
89}

91pub(crate) fn execute_instruction<M, I, V, Ty, Info, S>(
92    module: &LocalModule<I, V, Ty, Info>,
93    machine: &mut M,
94    instruction_override: Option<&I>,
95) -> eyre::Result<()>
96where
97    M: HasProgram<Program = ExecutionState<S>>
98        + Dispatch<LocalModule<I, V, Ty, Info>, Instruction = I>,
99{
100    let program = machine.program_mut();
101    program.ensure_running()?;
102    let position = program.pc;
103    let fetched = module
104        .code
105        .get(position as usize)
106        .ok_or_else(|| instruction_out_of_bounds(position, module.code.len()))?;
107    let instruction = instruction_override.unwrap_or(fetched);
108    program.pc = position
109        .checked_add(1)
110        .ok_or_else(|| eyre::eyre!("program counter overflow"))?;
111    let result = machine.dispatch(instruction, module);
112    let program = machine.program_mut();
113    if result.is_err() || !matches!(program.run_state, RunState::Runnable) {
114        program.pc = position;
115    }
116    result
117}
