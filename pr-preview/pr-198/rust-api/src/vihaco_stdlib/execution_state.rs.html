4//! Checked invocation state and execution of an independently owned module.
5//!
6//! The host owns program data. `ExecutionState` owns frames, locals, operands,
7//! the private PC, and execution flags. A runner borrows both the module and the
8//! machine; instructions borrow the module while authored Steps mutate devices.
9//! Component authors choose messages/effects, without implementing stack logic.
10//! Custom drivers can wrap [`run`] and [`execute_instruction`] to change
11//! initialization, instruction selection, and when control returns to the host.
12//!
13//! ```
14//! use vihaco_abi::{Type, Value};
15//! use vihaco_module::module::{LocalModule, FunctionInfo, Signature};
16//! use vihaco_runtime::{composite, field, Apply, BuildMessage, Execute, HasInstructionSet, Step};
17//! use vihaco_stdlib::{ExecutionState, Image, execution_state::execute, program::program};
18//! use vihaco_cpu::{CPU, cpu};
19//!
20//! #[composite]
21//! struct Machine {
22//!     #[device(0)] #[program] execution: vihaco_stdlib::ExecutionState,
23//!     #[device(1)] cpu: vihaco_cpu::CPU,
24//! }
25//! type Instruction = <Machine as HasInstructionSet>::Runtime;
26//! type Program = LocalModule<Instruction, Value, Type>;
27//!
28//! impl<I> Step<field!(Machine::execution), I, Program> for Machine
29//! where
30//!     ExecutionState: Execute<I, Effect = (), Error = eyre::Report>,
31//!     Self: BuildMessage<field!(Machine::execution), I, Program>,
32//! {
33//!     fn step(&mut self, instruction: &I, module: &Program) -> eyre::Result<()> {
34//!         let message = <Self as BuildMessage<field!(Machine::execution), I, Program>>
35//!             ::build_message(self, instruction, module)?;
36//!         self.execution.execute(instruction, message)
37//!     }
38//! }
39//! impl<I> Step<field!(Machine::cpu), I, Program> for Machine
40//! where
41//!     CPU: Execute<I, Error = eyre::Report>,
42//!     Self: BuildMessage<field!(Machine::cpu), I, Program>,
43//!     ExecutionState: Apply<<CPU as Execute<I>>::Effect>,
44//! {
45//!     fn step(&mut self, instruction: &I, module: &Program) -> eyre::Result<()> {
46//!         let message = <Self as BuildMessage<field!(Machine::cpu), I, Program>>
47//!             ::build_message(self, instruction, module)?;
48//!         let effect = self.cpu.execute(instruction, message)?;
49//!         self.execution.apply(effect);
50//!         Ok(())
51//!     }
52//! }
53//!
54//! let module: Program = LocalModule {
55//!     code: vec![
56//!         Instruction::V_execution__program__ConstU64(program::ConstU64(20)),
57//!         Instruction::V_execution__program__ConstU64(program::ConstU64(22)),
58//!         Instruction::V_cpu__cpu__AddU64(cpu::AddU64),
59//!         Instruction::V_execution__program__Return(program::Return(1)),
60//!     ],
61//!     main_function: Some(0),
62//!     functions: vec![FunctionInfo {
63//!         name: 0, signature: Signature { params: vec![], ret: vec![Type::U64] },
64//!         local_count: 0, start_address: 0, end_address: 4, file: 0,
65//!     }],
66//!     ..Default::default()
67//! };
68//! // An ordinary outer owner keeps immutable program data beside its machine.
69//! struct Loaded {
70//!     image: Image<Instruction>,
71//!     machine: Machine,
72//! }
73//! impl Loaded {
74//!     fn run(&mut self) -> eyre::Result<()> {
75//!         execute(&self.image.module, &mut self.machine, &[])
76//!     }
77//! }
78//! let mut loaded = Loaded {
79//!     image: Image::new(module),
80//!     machine: Machine { execution: ExecutionState::new(), cpu: CPU::default() },
81//! };
82//! loaded.run()?;
83//! assert_eq!(loaded.machine.execution.stack(), &[42]);
84//! # Ok::<(), eyre::Report>(())
85//! ```

97/// No custom execution state is available. By using this for the
98/// default in [`RunState`], we can get exhaustive pattern matching
99/// for `Runnable` and `Halted` by default.
100#[derive(Debug, Clone, Copy, PartialEq, Eq)]
101pub enum NoCustomState {}
102
103/// Whether an invocation can execute its next instruction.
104#[derive(Debug, Clone, Copy, PartialEq, Eq)]
105pub enum RunState<S = NoCustomState> {
106    Runnable,
107    Halted,
108    Other(S),
109}
110
111#[vihaco_runtime::component]
112#[instructions { program }]
113/// Frames, locals, operands, and private execution state for one invocation.
114///
115/// The module is supplied by reference at entry and is never stored here.
116#[derive(Debug)]
117pub struct ExecutionState<S = NoCustomState> {
118    stack: Vec<Word>,
119    frames: Vec<Frame>,
120    pc: u32,
121    run_state: RunState<S>,
122    executing: bool,
123}
124
125impl<S> Default for ExecutionState<S> {
126    fn default() -> Self {
127        Self {
128            stack: Vec::new(),
129            frames: Vec::new(),
130            pc: 0,
131            run_state: RunState::Halted,
132            executing: false,
133        }
134    }
135}
136
137impl ExecutionState {
138    pub fn new() -> Self {
139        Self::default()
140    }
141}
142
143impl<S: Clone> Clone for ExecutionState<S> {
144    fn clone(&self) -> Self {
145        Self {
146            stack: self.stack.clone(),
147            frames: self.frames.clone(),
148            pc: self.pc,
149            run_state: self.run_state.clone(),
150            executing: false,
151        }
152    }
153}

161fn function_entry<I, V, Ty, Info>(
162    module: &LocalModule<I, V, Ty, Info>,
163    index: usize,
164) -> eyre::Result<(usize, usize, u32)> {
165    let function = module
166        .functions
167        .get(index)
168        .ok_or_else(|| unknown_function(index))?;
169    if (function.local_count as usize) < function.signature.params.len() {
170        return Err(invalid_local_count());
171    }
172    if (function.start_address as usize) >= module.code.len() {
173        return Err(invalid_call_target());
174    }
175    Ok((
176        function.signature.params.len(),
177        function.local_count as usize,
178        function.start_address,
179    ))
180}

182#[cold]
183#[inline(never)]
184fn unknown_function(index: usize) -> eyre::Report {
185    eyre::eyre!("unknown function reference {index}")
186}
