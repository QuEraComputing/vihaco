4//! Types as written in source, where structs and named types may be named.
5//!
6//! [`SourceType`] accepts every [`Type`] form plus `struct @Point` and
7//! `type @location`, at any nesting depth (`array struct @Point`). Parse a
8//! header and function signatures with it, then resolve the names against the
9//! section's own tables with [`ProgramMetadata::resolve_types`]; the
10//! signature check during lowering resolves function headers the same way
11//! (through [`ResolveType`]).
12

24/// A type as written in source: any [`Type`], where struct and named types may
25/// also be named (`struct @Point`, `type @location`), at any nesting depth
26/// (`array array struct @Point`).
27///
28/// ```
29/// # use vihaco_module::module::SourceType;
30/// # use vihaco_abi::traits::FromText;
31/// use vihaco_parser::Parse as _;
32/// use chumsky::Parser as _;
33/// let ty = SourceType::parser().parse("array struct @Point").into_result().unwrap();
34/// assert_eq!(ty.to_string(), "array struct @Point");
35/// ```
36#[derive(Clone, Debug, PartialEq, Eq, Hash)]
37pub enum SourceType {
38    /// A type with no names to resolve, such as `f64` or `array struct 0`.
39    Resolved(Type),
40    /// `array <elem>`, where the element type contains names.
41    Array(Box<SourceType>),
42    /// `struct @Name`.
43    StructSymbol(Ident),
44    /// `type @name`.
45    NamedSymbol(Ident),
46}
47
48impl From<Type> for SourceType {
49    fn from(ty: Type) -> Self {
50        SourceType::Resolved(ty)
51    }
52}
53
54impl<'src> Parse<'src> for SourceType {
55    fn parser() -> impl Parser<'src, &'src str, Self, extra::Err<Simple<'src, char>>> {
56        // `recursive` needs cloneable parsers; `Parse::parser` is not `Clone`.
57        let resolved = Type::parser().boxed();
58        recursive(move |source| {
59            let named_struct = just("struct")
60                .then(text::whitespace().at_least(1))
61                .then(just('@'))
62                .ignore_then(vihaco_parser::ident())
63                .map(|name| SourceType::StructSymbol(Ident(name)));
64            let named_type = just("type")
65                .then(text::whitespace().at_least(1))
66                .then(just('@'))
67                .ignore_then(vihaco_parser::ident())
68                .map(|name| SourceType::NamedSymbol(Ident(name)));
69            let array = just("array")
70                .then(text::whitespace().at_least(1))
71                .ignore_then(source)
72                .map(|elem| match elem {
73                    // Keep fully resolved types in one piece.
74                    SourceType::Resolved(elem) => SourceType::Resolved(Type::array(elem)),
75                    elem => SourceType::Array(Box::new(elem)),
76                });
77            choice((
78                named_struct,
79                named_type,
80                array,
81                resolved.clone().map(SourceType::Resolved),
82            ))
83        })
84    }
85}
86
87impl fmt::Display for SourceType {
88    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
89        match self {
90            SourceType::Resolved(ty) => write!(f, "{ty}"),
91            SourceType::Array(elem) => write!(f, "array {elem}"),
92            SourceType::StructSymbol(name) => write!(f, "struct @{}", name.as_str()),
93            SourceType::NamedSymbol(name) => write!(f, "type @{}", name.as_str()),
94        }
95    }
96}

98/// Struct and named-type names of a section, for resolving [`SourceType`]s.
99#[derive(Debug, Default)]
100pub struct TypeLookup<'a> {
101    structs: HashMap<&'a str, u32>,
102    types: HashMap<&'a str, u32>,
103}
104
105impl<'a> TypeLookup<'a> {
106    /// The names of a section's struct and named-type tables. Entries whose
107    /// name is not a string-table entry are skipped, and the first of
108    /// duplicate names wins; metadata validation reports both.
109    pub fn new<Ty>(
110        structs: &[StructInfo<Ty>],
111        types: &[NamedTypeInfo<Ty>],
112        strings: &'a [String],
113    ) -> Self {
114        let mut lookup = Self::default();
115        for (index, info) in structs.iter().enumerate() {
116            if let Some(name) = strings.get(info.name as usize) {
117                lookup.structs.entry(name.as_str()).or_insert(index as u32);
118            }
119        }
120        for (index, info) in types.iter().enumerate() {
121            if let Some(name) = strings.get(info.name as usize) {
122                lookup.types.entry(name.as_str()).or_insert(index as u32);
123            }
124        }
125        lookup
126    }
127
128    /// The struct-table index of the struct named `name`.
129    pub fn struct_index(&self, name: &str) -> Option<u32> {
130        self.structs.get(name).copied()
131    }
132
133    /// The named-type-table index of the named type called `name`.
134    pub fn named_type_index(&self, name: &str) -> Option<u32> {
135        self.types.get(name).copied()
136    }
137}

139/// A source type that resolves to the type model `T` against a section's
140/// tables.
141///
142/// Every type resolves to itself, so code generic over the source type
143/// accepts both plain [`Type`]s and [`SourceType`]s.
144pub trait ResolveType<T> {
145    /// Resolve the names in this type.
146    ///
147    /// # Errors
148    /// A name with no table entry: `undefined struct `@Pointt``.
149    fn resolve_type(&self, lookup: &TypeLookup<'_>) -> eyre::Result<T>;
150}
151
152impl<T: Clone> ResolveType<T> for T {
153    fn resolve_type(&self, _: &TypeLookup<'_>) -> eyre::Result<T> {
154        Ok(self.clone())
155    }
156}
157
158impl ResolveType<Type> for SourceType {
159    fn resolve_type(&self, lookup: &TypeLookup<'_>) -> eyre::Result<Type> {
160        Ok(match self {
161            SourceType::Resolved(ty) => ty.clone(),
162            SourceType::Array(elem) => {
163                Type::array(ResolveType::<Type>::resolve_type(&**elem, lookup)?)
164            }
165            SourceType::StructSymbol(name) => Type::Struct(
166                lookup
167                    .struct_index(name.as_str())
168                    .ok_or_else(|| eyre::eyre!("undefined struct `@{}`", name.as_str()))?,
169            ),
170            SourceType::NamedSymbol(name) => Type::Named(
171                lookup
172                    .named_type_index(name.as_str())
173                    .ok_or_else(|| eyre::eyre!("undefined named type `@{}`", name.as_str()))?,
174            ),
175        })
176    }
177}

179#[cfg(test)]
180mod tests {
181    use super::*;
182    use crate::module::{FieldInfo, FunctionInfo, Parameter, ProgramMetadata, Signature};
183    use vihaco_abi::Value;
184    use vihaco_abi::traits::FromText;
185
186    fn parse(text: &str) -> SourceType {
187        SourceType::parser().parse(text).into_result().unwrap()
188    }
189
190    #[test]
191    fn source_types_accept_names_at_any_depth() {
192        assert_eq!(parse("f64"), SourceType::Resolved(Type::F64));
193        assert_eq!(
194            parse("array struct 0"),
195            SourceType::Resolved(Type::array(Type::Struct(0)))
196        );
197        let nested = parse("array array struct @Point");
198        assert_eq!(nested.to_string(), "array array struct @Point");
199        assert_eq!(parse("type @location").to_string(), "type @location");
200    }
201
202    #[test]
203    fn headers_resolve_names_against_their_own_tables() {
204        let text = "metadata {
205            functions: [{ name: 0, signature: { params: [{ name: 1, ty: struct @Point }],
206                                                ret: [type @location] },
207                          locals: [struct @Point, array type @location],
208                          start_address: 0, end_address: 1, file: 0 }],
209            structs: [{ name: 2, fields: [{ name: 3, ty: array struct @Point }] }],
210            types: [{ name: 4, repr: u64 }],
211            labels: [], constants: [],
212            strings: [\"main\", \"p0\", \"Point\", \"next\", \"location\"],
213            main_function: 0, source_symbols: [], extra: {},
214        }";
215        let metadata = ProgramMetadata::<Value, SourceType>::from_text(text)
216            .unwrap()
217            .resolve_types::<Type>()
218            .unwrap();
219        let expected = FunctionInfo {
220            name: 0,
221            signature: Signature {
222                params: vec![Parameter {
223                    name: 1,
224                    ty: Type::Struct(0),
225                }],
226                ret: vec![Type::Named(0)],
227            },
228            locals: vec![Type::Struct(0), Type::array(Type::Named(0))],
229            start_address: 0,
230            end_address: 1,
231            file: 0,
232        };
233        assert_eq!(metadata.functions, [expected]);
234        assert_eq!(
235            metadata.structs[0].fields,
236            [FieldInfo {
237                name: 3,
238                ty: Type::array(Type::Struct(0))
239            }]
240        );
241        metadata.into_module::<()>(vec![()]).unwrap();
242
243        let typo = text.replace("locals: [struct @Point", "locals: [struct @Pointt");
244        let err = ProgramMetadata::<Value, SourceType>::from_text(&typo)
245            .unwrap()
246            .resolve_types::<Type>()
247            .unwrap_err();
248        assert_eq!(
249            err.to_string(),
250            "metadata.functions[0].locals[0]: undefined struct `@Pointt`"
251        );
252    }
