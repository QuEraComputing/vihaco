51#[derive(Debug, Clone, PartialEq, vihaco_parser_derive::Parse)]
52#[syntax_class(metadata, open = "{", close = "}")]
53pub struct FunctionInfo<Type> {
54    pub name: u32, // index into the string interner
55    pub signature: Signature<Type>,
56    /// The declared type of every local slot, parameters first: the leading
57    /// `signature.params.len()` entries are the parameter types. Written
58    /// `locals: [u64, f64]` in metadata headers. At most [`MAX_LOCALS`]
59    /// entries, checked by [`ProgramMetadata::validate`].
60    pub locals: Vec<Type>,
61    pub start_address: u32, // corresponds to a label noop
62    pub end_address: u32,   // corresponds to a label noop
63    pub file: u32,
64}
65
66/// The largest number of local slots, parameters included, a function may
67/// declare. [`ProgramMetadata::validate`] enforces it, and runtime frame entry
68/// re-checks it before allocating, since modules can also be built directly.
69pub const MAX_LOCALS: usize = 65_536;
70
71/// The default limit on nested call frames (the entry function's frame
72/// included) for the standard runtimes: the stdlib `ExecutionState` and the
73/// CPU. A call that would exceed it fails, naming the callee, instead of
74/// recursing until memory runs out.
75///
76/// Verification cannot bound recursion, so this is a runtime check. Each frame
77/// holds at most [`MAX_LOCALS`] words, so the stack is bounded by
78/// `limit × MAX_LOCALS` words plus operands; embedders that need a tighter
79/// memory bound lower the limit (`set_max_call_depth`).
80pub const DEFAULT_MAX_CALL_DEPTH: usize = 10_000;
81
82impl<Type> FunctionInfo<Type> {
83    /// The number of local slots, parameters included (`locals.len()`).
84    pub fn local_count(&self) -> usize {
85        self.locals.len()
86    }
87
88    /// Check the frame this function declares: at most [`MAX_LOCALS`]
89    /// locals, at least one per parameter, and leading locals equal to the
90    /// parameter types.
91    ///
92    /// [`ProgramMetadata::validate`] reports these with the metadata path;
93    /// the verifier repeats the check for modules built without validation.
94    pub fn check_frame(&self) -> Result<(), FrameError<Type>>
95    where
96        Type: PartialEq + Clone,
97    {
98        let (locals, params) = (&self.locals, &self.signature.params);
99        if locals.len() > MAX_LOCALS {
100            return Err(FrameError::TooManyLocals {
101                count: locals.len(),
102            });
103        }
104        if locals.len() < params.len() {
105            return Err(FrameError::FewerLocalsThanParameters {
106                locals: locals.len(),
107                parameters: params.len(),
108            });
109        }
110        match locals
111            .iter()
112            .zip(params)
113            .enumerate()
114            .find(|(_, (local, param))| **local != param.ty)
115        {
116            Some((index, (local, param))) => Err(FrameError::LocalDisagreesWithParameter {
117                index,
118                local: local.clone(),
119                parameter: param.ty.clone(),
120            }),
121            None => Ok(()),
122        }
123    }
124}

126/// A function's declared frame is invalid; see [`FunctionInfo::check_frame`].
127#[derive(Debug, Clone, PartialEq, Eq, Hash)]
128pub enum FrameError<Ty> {
129    /// More than [`MAX_LOCALS`] locals.
130    TooManyLocals {
131        /// The number of locals declared.
132        count: usize,
133    },
134    /// Fewer locals than parameters: every parameter occupies a local.
135    FewerLocalsThanParameters {
136        /// The number of locals declared.
137        locals: usize,
138        /// The number of parameters in the signature.
139        parameters: usize,
140    },
141    /// A leading local's type differs from its parameter's.
142    LocalDisagreesWithParameter {
143        /// The local and parameter index.
144        index: usize,
145        /// The local's declared type.
146        local: Ty,
147        /// The parameter's declared type.
148        parameter: Ty,
149    },
150}
151
152impl<Ty> FrameError<Ty> {
153    /// Display the error, naming parameters with `parameter_name` (for
154    /// example from the string table).
155    pub fn display_with<'a>(
156        &'a self,
157        parameter_name: impl Fn(usize) -> Option<&'a str> + 'a,
158        ty: impl Fn(&'a Ty) -> String + 'a,
159    ) -> impl std::fmt::Display + 'a {
160        FrameErrorDisplay {
161            error: self,
162            parameter_name: Box::new(parameter_name),
163            ty: Box::new(ty),
164        }
165    }
166}

168type NameFn<'a> = Box<dyn Fn(usize) -> Option<&'a str> + 'a>;
169
170struct FrameErrorDisplay<'a, Ty> {
171    error: &'a FrameError<Ty>,
172    parameter_name: NameFn<'a>,
173    ty: Box<dyn Fn(&'a Ty) -> String + 'a>,
174}
175
176impl<Ty> std::fmt::Display for FrameErrorDisplay<'_, Ty> {
177    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
178        match self.error {
179            FrameError::TooManyLocals { count } => {
180                write!(f, "{count} locals exceed the maximum of {MAX_LOCALS}")
181            }
182            FrameError::FewerLocalsThanParameters { locals, parameters } => write!(
183                f,
184                "{locals} local{} declared, but the signature has {parameters} parameter{} \
185                 (locals start with one per parameter)",
186                if *locals == 1 { "" } else { "s" },
187                if *parameters == 1 { "" } else { "s" },
188            ),
189            FrameError::LocalDisagreesWithParameter {
190                index,
191                local,
192                parameter,
193            } => {
194                write!(
195                    f,
196                    "local {index} is declared {}, but parameter {index}",
197                    (self.ty)(local)
198                )?;
199                if let Some(name) = (self.parameter_name)(*index) {
200                    write!(f, " ({name})")?;
201                }
202                write!(
203                    f,
204                    " is {} (locals start with the parameter types)",
205                    (self.ty)(parameter)
206                )
207            }
208        }
209    }
210}

223    pub params: Vec<Parameter<Ty>>,

225}

227impl<Ty> Signature<Ty> {
228    /// Whether this signature has exactly the parameter and result types of
229    /// `ty`, ignoring parameter names; `call_indirect` checks its target this
230    /// way before calling.
231    pub fn matches(&self, ty: &vihaco_abi::FunctionType) -> bool
232    where
233        Ty: PartialEq<vihaco_abi::Type>,
234    {
235        self.params.len() == ty.params.len()
236            && self.ret.len() == ty.results.len()
237            && self.params.iter().zip(&ty.params).all(|(p, t)| p.ty == *t)
238            && self.ret.iter().zip(&ty.results).all(|(r, t)| r == t)
239    }
240}

245        vihaco_abi::FunctionType {
246            params: self.params.iter().map(|param| param.ty.clone()).collect(),
247            results: self.ret.clone(),

249    }

276/// A named type declaration: a nominal type represented by a scalar word.
277/// `Type::Named(i)` (written `type i`) refers to entry `i` of the named-type
278/// table. At runtime a value of a named type is the same word as its `repr`;
279/// the name exists only for verification, where `type i` satisfies only
280/// itself.
281#[derive(Debug, Clone, PartialEq, vihaco_parser_derive::Parse)]
282#[syntax_class(metadata, open = "{", close = "}")]
283pub struct NamedTypeInfo<Ty> {
284    pub name: u32, // index into the string interner
285    /// The scalar word type representing values of this type: one of `i32`,
286    /// `i64`, `u32`, `u64`, `f32`, `f64` or `bool`.
287    pub repr: Ty,
288}

290/// Exposes the table references carried by a type, so module assembly can
291/// check `Type::Struct` indices against the struct table and `Type::Named`
292/// indices against the named-type table, and check named-type
293/// representations.
294pub trait StructTypeRef {
295    /// Visit every struct-table index this type refers to, including ones
296    /// nested inside compound types such as array element types.
297    fn visit_struct_refs(&self, visit: &mut dyn FnMut(u32));
298
299    /// Visit every named-type-table index this type refers to, including ones
300    /// nested inside compound types. Type models without named types keep the
301    /// default, which visits nothing.
302    fn visit_named_refs(&self, visit: &mut dyn FnMut(u32)) {
303        let _ = visit;
304    }
305
306    /// Whether this type may represent a named type (a scalar word type).
307    /// Type models without named types keep the default, `false`, so their
308    /// named-type tables must be empty.
309    fn is_named_type_repr(&self) -> bool {
310        false
311    }
312
313    /// Write this type for a message, with struct and named types shown by
314    /// the names in `names`. The default writes its `Display` form.
315    fn write_named(
316        &self,
317        f: &mut std::fmt::Formatter<'_>,
318        names: &dyn vihaco_abi::names::TypeNames,
319    ) -> std::fmt::Result
320    where
321        Self: std::fmt::Display,
322    {
323        let _ = names;
324        write!(f, "{self}")
325    }
326}
327
328impl StructTypeRef for vihaco_abi::Type {
329    fn visit_struct_refs(&self, visit: &mut dyn FnMut(u32)) {
330        vihaco_abi::Type::visit_struct_refs(self, visit)
331    }
332
333    fn visit_named_refs(&self, visit: &mut dyn FnMut(u32)) {
334        vihaco_abi::Type::visit_named_refs(self, visit)
335    }
336
337    fn is_named_type_repr(&self) -> bool {
338        self.is_scalar_word()
339    }
340
341    fn write_named(
342        &self,
343        f: &mut std::fmt::Formatter<'_>,
344        names: &dyn vihaco_abi::names::TypeNames,
345    ) -> std::fmt::Result {
346        write!(f, "{}", self.display_with(names))
347    }
348}

366struct SignatureType<'a, Ty> {
367    signature: &'a Signature<Ty>,
368    names: &'a dyn vihaco_abi::names::TypeNames,
369}

371impl<Ty: StructTypeRef + std::fmt::Display> std::fmt::Display for SignatureType<'_, Ty> {
372    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
373        let list = |f: &mut std::fmt::Formatter<'_>, types: &mut dyn Iterator<Item = &Ty>| {
374            f.write_str("(")?;
375            for (index, ty) in types.enumerate() {
376                if index > 0 {
377                    f.write_str(", ")?;
378                }
379                ty.write_named(f, self.names)?;
380            }
381            f.write_str(")")
382        };
383        list(f, &mut self.signature.params.iter().map(|param| &param.ty))?;
384        f.write_str(" -> ")?;
385        match self.signature.ret.as_slice() {
386            [single] => single.write_named(f, self.names),
387            results => list(f, &mut results.iter()),
388        }
389    }
390}

410impl<I, V, Ty, Info> std::fmt::Display for LocalModule<I, V, Ty, Info>
411where
412    I: std::fmt::Display,
413    V: std::fmt::Display,
414    Ty: std::fmt::Display,
415    Info: std::fmt::Display,
416{
417    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
418        ".text".keyword().fmt(f)?;
419        writeln!(f)?;
420
421        for (addr, inst) in self.code.iter().enumerate() {
422            write!(f, "0x{:04X} ", addr)?;
423            inst.fmt(f)?;
424            writeln!(f)?;
425        }
426
427        writeln!(f)?;
428        ".const".keyword().fmt(f)?;
429        writeln!(f)?;
430        for (const_idx, constant) in self.constants.iter().enumerate() {
431            write!(f, "0x{:04X} ", const_idx)?;
432            constant.fmt(f)?;
433            writeln!(f)?;
434        }
435
436        writeln!(f)?;
437        ".string".keyword().fmt(f)?;
438        writeln!(f)?;
439        for (string_idx, string) in self.strings.iter().enumerate() {
440            if string_idx > 0 {
441                writeln!(f)?;
442            }
443            write!(f, "0x{:04X} \"{}\"", string_idx, string.escape_default())?;
444        }
445
446        writeln!(f)?;
447        if !self.structs.is_empty() {
448            writeln!(f)?;
449            ".struct".keyword().fmt(f)?;
450            writeln!(f)?;
451            let name = |index: u32| {
452                self.strings
453                    .get(index as usize)
454                    .map_or_else(|| format!("#{index}"), Clone::clone)
455            };
456            for (struct_idx, info) in self.structs.iter().enumerate() {
457                write!(f, "0x{:04X} {} {{", struct_idx, name(info.name))?;
458                for (field_idx, field) in info.fields.iter().enumerate() {
459                    let sep = if field_idx == 0 { " " } else { ", " };
460                    write!(f, "{sep}{}: {}", name(field.name), field.ty)?;
461                }
462                let close = if info.fields.is_empty() { "}" } else { " }" };
463                writeln!(f, "{close}")?;
464            }
465        }
466        if !self.types.is_empty() {
467            writeln!(f)?;
468            ".type".keyword().fmt(f)?;
469            writeln!(f)?;
470            for (type_idx, info) in self.types.iter().enumerate() {
471                let name = self
472                    .strings
473                    .get(info.name as usize)
474                    .map_or_else(|| format!("#{}", info.name), Clone::clone);
475                writeln!(f, "0x{:04X} {} = {}", type_idx, name, info.repr)?;
476            }
477        }
478        if !self.source_symbols.is_empty() {
479            ".machine".keyword().fmt(f)?;
480            writeln!(f, " symbols {{")?;
481            for symbol in &self.source_symbols {
482                writeln!(f, "  {}: {},", symbol.index, symbol.name)?;
483            }
484            writeln!(f, "}}")?;
485            writeln!(f)?;
486        }
487        writeln!(f)?;
488        ".feature".keyword().fmt(f)?;
489        writeln!(f)?;
490        self.extra.fmt(f)?;
491        Ok(())
492    }
493}
