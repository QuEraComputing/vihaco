4//! Checked invocation state and execution of an independently owned module.
5//!
6//! The host owns program data. `ExecutionState` owns frames, locals, operands,
7//! the private PC, and execution flags. A runner borrows both the module and the
8//! machine; instructions borrow the module while authored Steps mutate devices.
9//! Component authors choose messages/effects, without implementing stack logic.
10//!
11//! ```
12//! use vihaco_abi::{Type, Value};
13//! use vihaco_module::module::{LocalModule, FunctionInfo, Signature};
14//! use vihaco_runtime::{composite, field, Apply, BuildMessage, Execute, HasInstructionSet, Step};
15//! use vihaco_stdlib::{ExecutionState, Image, Verified, execution_state::execute, program::program};
16//! use vihaco_cpu::{CPU, cpu};
17//!
18//! #[composite]
19//! struct Machine {
20//!     #[device(0)] #[program] execution: vihaco_stdlib::ExecutionState,
21//!     #[device(1)] cpu: vihaco_cpu::CPU,
22//! }
23//! type Instruction = <Machine as HasInstructionSet>::Runtime;
24//! type Program = LocalModule<Instruction, Value, Type>;
25//!
26//! impl<I> Step<field!(Machine::execution), I, Program> for Machine
27//! where
28//!     ExecutionState: Execute<I, Effect = (), Error = eyre::Report>,
29//!     Self: BuildMessage<field!(Machine::execution), I, Program>,
30//! {
31//!     fn step(&mut self, instruction: &I, module: &Program) -> eyre::Result<()> {
32//!         let message = <Self as BuildMessage<field!(Machine::execution), I, Program>>
33//!             ::build_message(self, instruction, module)?;
34//!         self.execution.execute(instruction, message)
35//!     }
36//! }
37//! impl<I> Step<field!(Machine::cpu), I, Program> for Machine
38//! where
39//!     CPU: Execute<I, Error = eyre::Report>,
40//!     Self: BuildMessage<field!(Machine::cpu), I, Program>,
41//!     ExecutionState: Apply<<CPU as Execute<I>>::Effect>,
42//! {
43//!     fn step(&mut self, instruction: &I, module: &Program) -> eyre::Result<()> {
44//!         let message = <Self as BuildMessage<field!(Machine::cpu), I, Program>>
45//!             ::build_message(self, instruction, module)?;
46//!         let effect = self.cpu.execute(instruction, message)?;
47//!         self.execution.apply(effect);
48//!         Ok(())
49//!     }
50//! }
51//!
52//! let module: Program = LocalModule {
53//!     code: vec![
54//!         Instruction::V_execution__program__ConstU64(program::ConstU64(20)),
55//!         Instruction::V_execution__program__ConstU64(program::ConstU64(22)),
56//!         Instruction::V_cpu__cpu__AddU64(cpu::AddU64),
57//!         Instruction::V_execution__program__Return(program::Return(1)),
58//!     ],
59//!     main_function: Some(0),
60//!     structs: vec![],
61//!     types: vec![],
62//!     functions: vec![FunctionInfo {
63//!         name: 0, signature: Signature { params: vec![], ret: vec![Type::U64] },
64//!         locals: vec![], start_address: 0, end_address: 4, file: 0,
65//!     }],
66//!     ..Default::default()
67//! };
68//! // An ordinary outer owner keeps immutable program data beside its machine.
69//! struct Loaded {
70//!     image: Image<Instruction>,
71//!     machine: Machine,
72//! }
73//! impl Loaded {
74//!     fn run(&mut self) -> eyre::Result<()> {
75//!         execute(self.image.module(), &mut self.machine, &[])
76//!     }
77//! }
78//! let mut loaded = Loaded {
79//!     image: Image::new(Verified::new(module)?),
80//!     machine: Machine { execution: ExecutionState::new(), cpu: CPU::default() },
81//! };
82//! loaded.run()?;
83//! assert_eq!(loaded.machine.execution.stack(), &[42]);
84//! # Ok::<(), eyre::Report>(())
85//! ```

99/// Frames, locals, operands, and private execution state for one invocation.
100///
101/// The module is supplied by reference at entry and is never stored here.
102#[derive(Debug)]
103pub struct ExecutionState {
104    stack: Vec<Word>,
105    frames: Vec<Frame>,
106    pc: u32,
107    halted: bool,
108    executing: bool,
109    max_call_depth: usize,
110}
111
112impl Default for ExecutionState {
113    fn default() -> Self {
114        Self {
115            stack: Vec::new(),
116            frames: Vec::new(),
117            pc: 0,
118            halted: true,
119            executing: false,
120            max_call_depth: DEFAULT_MAX_CALL_DEPTH,
121        }
122    }
123}
124
125impl ExecutionState {
126    pub fn new() -> Self {
127        Self::default()
128    }
129
130    /// The most call frames (the entry function's included) an invocation
131    /// may have at once; [`DEFAULT_MAX_CALL_DEPTH`] unless changed.
132    pub fn max_call_depth(&self) -> usize {
133        self.max_call_depth
134    }
135
136    /// Limit nested call frames to `depth`. A call that would exceed it fails
137    /// with an error naming the callee, before any frame storage is reserved.
138    pub fn set_max_call_depth(&mut self, depth: usize) {
139        self.max_call_depth = depth;
140    }
141
142    /// [`Self::set_max_call_depth`], by value.
143    pub fn with_max_call_depth(mut self, depth: usize) -> Self {
144        self.set_max_call_depth(depth);
145        self
146    }
147
148    /// Fail if entering `function` would exceed the call-depth limit.
149    fn check_call_depth<I, V, Ty, Info>(
150        &self,
151        module: &LocalModule<I, V, Ty, Info>,
152        function: usize,
153    ) -> eyre::Result<()> {
154        if self.frames.len() >= self.max_call_depth {
155            return Err(call_too_deep(module, function, self.max_call_depth));
156        }
157        Ok(())
158    }
159}
160
161impl Clone for ExecutionState {
162    fn clone(&self) -> Self {
163        Self {
164            stack: self.stack.clone(),
165            frames: self.frames.clone(),
166            pc: self.pc,
167            halted: self.halted,
168            executing: false,
169            max_call_depth: self.max_call_depth,
170        }
171    }
172}

180fn function_entry<I, V, Ty, Info>(
181    module: &LocalModule<I, V, Ty, Info>,
182    index: usize,
183) -> eyre::Result<(usize, usize, u32)> {
184    let function = module
185        .functions
186        .get(index)
187        .ok_or_else(|| unknown_function(index))?;
188    let locals = function.local_count();
189    // Modules can be built without `ProgramMetadata::validate`, so re-check
190    // the frame bound before any caller reserves or resizes storage.
191    if locals > MAX_LOCALS {
192        return Err(too_many_locals(module, index, locals));
193    }
194    if locals < function.signature.params.len() {
195        return Err(invalid_local_count());
196    }
197    if (function.start_address as usize) >= module.code.len() {
198        return Err(invalid_call_target());
199    }
200    Ok((
201        function.signature.params.len(),
202        locals,
203        function.start_address,
204    ))
205}

207#[cold]

209fn unknown_function(index: usize) -> eyre::Report {
210    eyre::eyre!("unknown function reference {index}")
211}

213/// `@name (function N)`, or `function N` if the name is not in the string table.
214fn describe_function<I, V, Ty, Info>(module: &LocalModule<I, V, Ty, Info>, index: usize) -> String {
215    let name = module
216        .functions
217        .get(index)
218        .and_then(|function| module.strings.get(function.name as usize));
219    match name {
220        Some(name) => format!("`@{name}` (function {index})"),
221        None => format!("function {index}"),
222    }
223}

225#[cold]
226#[inline(never)]
227fn too_many_locals<I, V, Ty, Info>(
228    module: &LocalModule<I, V, Ty, Info>,
229    index: usize,
230    locals: usize,
231) -> eyre::Report {
232    eyre::eyre!(
233        "{} declares {locals} locals, more than the maximum of {MAX_LOCALS}",
234        describe_function(module, index)
235    )
236}

238#[cold]
239#[inline(never)]
240fn call_too_deep<I, V, Ty, Info>(
241    module: &LocalModule<I, V, Ty, Info>,
242    index: usize,
243    limit: usize,
244) -> eyre::Report {
245    eyre::eyre!(
246        "call to {} exceeds the maximum call depth of {limit} frames",
247        describe_function(module, index)
248    )
249}
