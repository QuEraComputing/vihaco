11/// Enter main and execute through the composite's authored Steps.
12///
13/// This is the low-level, **unverified** runner: it accepts any plain module.
14/// Load-time verification is enforced by [`Image`](crate::Image), which only
15/// takes a `Verified` module, and so by [`Loadable`](crate::Loadable); use those
16/// for programs from untrusted sources. Runtime checks (operand bounds, heap
17/// tags, frame and call-depth limits) still apply here.
18///
19/// Devices remain in place. Instructions borrow the caller-owned module; there
20/// is no instruction clone, generated Step, or imposed effect-application policy.
21///
22/// # Errors
23/// Returns entry, checked-fetch, or authored Dispatch/Step errors. Operand and
24/// component mutations made before an error are retained.
25pub fn execute<M, I, V, Ty, Info>(
26    module: &LocalModule<I, V, Ty, Info>,
27    machine: &mut M,
28    arguments: &[Word],
29) -> eyre::Result<()>
30where
31    M: HasProgram<Program = ExecutionState>
32        + Dispatch<LocalModule<I, V, Ty, Info>, Instruction = I>,
33{
34    machine.program_mut().start(module, arguments)?;
35    resume(module, machine)
36}

38/// Continue an invocation until Halt/root return or an error.
39///
40/// Fetch establishes the sequential continuation before the Step runs. Program
41/// instructions can replace that continuation. A fault or terminal instruction
42/// leaves its original position available through [`ExecutionState::pc()`].
43///
44/// The caller must supply the same unchanged module used to start the active
45/// invocation. This checked API validates accesses, but does not enforce module
46/// identity. After an error, earlier operand consumption and effects remain;
47/// the caller is responsible for deciding whether the invocation can resume.
48///
49/// # Errors
50/// Rejects reentrant execution, invalid running state, invalid fetches, PC
51/// overflow, or a failure returned by the authored Dispatch/Step. A step's
52/// failure is a [`StepError`] located at its instruction (`at @main+2
53/// (address 2) `cpu::cpu::get_field @Point.y`: ...`), with heap types shown
54/// by the module's names; the instruction text comes from
55/// [`Dispatch::instruction_text`], so a hand-written machine without it shows
56/// the location alone.
57pub fn resume<M, I, V, Ty, Info>(
58    module: &LocalModule<I, V, Ty, Info>,
59    machine: &mut M,
60) -> eyre::Result<()>
61where
62    M: HasProgram<Program = ExecutionState>
63        + Dispatch<LocalModule<I, V, Ty, Info>, Instruction = I>,
64{
65    let program = machine.program_mut();
66    eyre::ensure!(
67        !program.executing,
68        "program execution is already in progress"
69    );
70    program.executing = true;
71    // Capture code once from the independent module borrow.
72    let code = &module.code;
73    let result = (|| {
74        while !machine.program().halted {
75            let program = machine.program_mut();
76            program.ensure_running()?;
77            let position = program.pc;
78            let instruction = code
79                .get(position as usize)
80                .ok_or_else(|| instruction_out_of_bounds(position, code.len()))?;
81            program.pc = position
82                .checked_add(1)
83                .ok_or_else(|| eyre::eyre!("program counter overflow"))?;
84            // `instruction` borrows the external module, never machine.
85            let result = machine.dispatch(instruction, module);
86            let program = machine.program_mut();
87            if result.is_err() || program.halted {
88                program.pc = position;
89            }
90            result.map_err(|error| {
91                let names = module.names();
92                let text = M::instruction_text(instruction, &names);
93                step_error(error, module, position, text)
94            })?;
95        }
96        Ok(())
97    })();
98    machine.program_mut().executing = false;
99    result
100}

102/// Locate a step's failure at its instruction, showing a heap error's types
103/// by the module's names.
104#[cold]
105#[inline(never)]
106fn step_error<I, V, Ty, Info>(
107    error: eyre::Report,
108    module: &LocalModule<I, V, Ty, Info>,
109    position: u32,
110    instruction: Option<String>,
111) -> eyre::Report {
112    let names = module.names();
113    let named = error
114        .downcast_ref::<HeapError>()
115        .map(|heap| NamedHeapError::new(heap.clone(), &names));
116    let location = match instruction {
117        Some(text) => format!("{} `{text}`", names.location(position)),
118        None => names.location(position).to_string(),
119    };
120    StepError {
121        address: position,
122        location,
123        named,
124        error,
125    }
126    .into()
127}

129/// An instruction failed while executing a program.
130///
131/// Like an eyre context layer, it displays only where the failure happened:
132/// the location and the instruction, `at @main+2 (address 2)
133/// `cpu::cpu::get_field @Point.x``. What went wrong is its
134/// [`source`](std::error::Error::source), so `{:#}` (or walking
135/// `eyre::Report::chain`) shows both, once each: `at @main+2 (address 2)
136/// `cpu::cpu::get_field @Point.x`: type error: heap object (slot 1,
137/// generation 1) is struct Pair, expected struct Point`.
138///
139/// The source is the step's own error (for example a device's error type),
140/// except that a [`HeapError`] appears as a [`NamedHeapError`], rendered with
141/// the module's names; its [`error`](NamedHeapError::error) is the
142/// structured `HeapError`. [`StepError::error`] returns the step's error
143/// unchanged. `downcast_ref` on the returned report itself finds the
144/// `StepError`: search the chain for the cause.
145#[derive(Debug)]
146pub struct StepError {
147    address: u32,
148    location: String,
149    named: Option<NamedHeapError>,
150    error: eyre::Report,
151}
152
153impl StepError {
154    /// The section-local address of the failing instruction.
155    pub fn address(&self) -> u32 {
156        self.address
157    }
158
159    /// The error the instruction's step returned.
160    pub fn error(&self) -> &eyre::Report {
161        &self.error
162    }
163
164    /// Consume the location, returning the step's error.
165    pub fn into_error(self) -> eyre::Report {
166        self.error
167    }
168}
169
170impl std::fmt::Display for StepError {
171    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
172        write!(f, "at {}", self.location)
173    }
174}
175
176impl std::error::Error for StepError {
177    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
178        match &self.named {
179            Some(named) => Some(named),
180            None => Some(self.error.as_ref()),
181        }
182    }
183}
