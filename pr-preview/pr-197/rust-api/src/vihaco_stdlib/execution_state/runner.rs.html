11/// Enter main and execute through the composite's authored Steps.
12///
13/// Devices remain in place. Instructions borrow the caller-owned module; there
14/// is no instruction clone, generated Step, or imposed effect-application policy.
15///
16/// # Errors
17/// Returns entry, checked-fetch, or authored Dispatch/Step errors. Operand and
18/// component mutations made before an error are retained.
19pub fn execute<M, I, V, Ty, Info>(
20    module: &LocalModule<I, V, Ty, Info>,
21    machine: &mut M,
22    arguments: &[Word],
23) -> eyre::Result<()>
24where
25    I: std::fmt::Display,
26    M: HasProgram<Program = ExecutionState>
27        + Dispatch<LocalModule<I, V, Ty, Info>, Instruction = I>,
28{
29    machine.program_mut().start(module, arguments)?;
30    resume(module, machine)
31}

33/// Continue an invocation until Halt/root return or an error.
34///
35/// Fetch establishes the sequential continuation before the Step runs. Program
36/// instructions can replace that continuation. A fault or terminal instruction
37/// leaves its original position available through [`ExecutionState::pc()`].
38///
39/// The caller must supply the same unchanged module used to start the active
40/// invocation. This checked API validates accesses, but does not enforce module
41/// identity. After an error, earlier operand consumption and effects remain;
42/// the caller is responsible for deciding whether the invocation can resume.
43///
44/// # Errors
45/// Rejects reentrant execution, invalid running state, invalid fetches, PC
46/// overflow, or a failure returned by the authored Dispatch/Step. A step's
47/// failure is located: `at @main+2 (address 2): cpu::cpu::get_field ...`,
48/// followed by the error, with heap types shown by the module's names.
49pub fn resume<M, I, V, Ty, Info>(
50    module: &LocalModule<I, V, Ty, Info>,
51    machine: &mut M,
52) -> eyre::Result<()>
53where
54    I: std::fmt::Display,
55    M: HasProgram<Program = ExecutionState>
56        + Dispatch<LocalModule<I, V, Ty, Info>, Instruction = I>,
57{
58    let program = machine.program_mut();
59    eyre::ensure!(
60        !program.executing,
61        "program execution is already in progress"
62    );
63    program.executing = true;
64    // Capture code once from the independent module borrow.
65    let code = &module.code;
66    let result = (|| {
67        while !machine.program().halted {
68            let program = machine.program_mut();
69            program.ensure_running()?;
70            let position = program.pc;
71            let instruction = code
72                .get(position as usize)
73                .ok_or_else(|| instruction_out_of_bounds(position, code.len()))?;
74            program.pc = position
75                .checked_add(1)
76                .ok_or_else(|| eyre::eyre!("program counter overflow"))?;
77            // `instruction` borrows the external module, never machine.
78            let result = machine.dispatch(instruction, module);
79            let program = machine.program_mut();
80            if result.is_err() || program.halted {
81                program.pc = position;
82            }
83            result.map_err(|error| step_error(error, module, position, instruction))?;
84        }
85        Ok(())
86    })();
87    machine.program_mut().executing = false;
88    result
89}

91/// Locate a step's failure at its instruction, showing a heap error's types
92/// by the module's names.
93#[cold]
94#[inline(never)]
95fn step_error<I: std::fmt::Display, V, Ty, Info>(
96    error: eyre::Report,
97    module: &LocalModule<I, V, Ty, Info>,
98    position: u32,
99    instruction: &I,
100) -> eyre::Report {
101    let names = module.names();
102    let message = match error.downcast_ref::<HeapError>() {
103        Some(heap) => heap.display_with(&names).to_string(),
104        None => format!("{error:#}"),
105    };
106    StepError {
107        address: position,
108        location: format!("{} `{instruction}`", names.location(position)),
109        message,
110        error,
111    }
112    .into()
113}

115/// An instruction failed while executing a program.
116///
117/// Displays as one line, the location, the instruction, then the failure:
118/// `at @main+2 (address 2) `cpu::cpu::get_field FieldSlot { struct_id: 0,
119/// slot: 0 }`: type error: heap object (slot 1, generation 1) is struct
120/// Pair, expected struct Point`. The original error is kept for inspection
121/// ([`StepError::error`]), for example to downcast a device's own error type.
122#[derive(Debug)]
123pub struct StepError {
124    address: u32,
125    location: String,
126    message: String,
127    error: eyre::Report,
128}
129
130impl StepError {
131    /// The section-local address of the failing instruction.
132    pub fn address(&self) -> u32 {
133        self.address
134    }
135
136    /// The error the instruction's step returned.
137    pub fn error(&self) -> &eyre::Report {
138        &self.error
139    }
140
141    /// Consume the location, returning the step's error.
142    pub fn into_error(self) -> eyre::Report {
143        self.error
144    }
145}
146
147impl std::fmt::Display for StepError {
148    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
149        write!(f, "at {}: {}", self.location, self.message)
150    }
151}

153impl std::error::Error for StepError {}
