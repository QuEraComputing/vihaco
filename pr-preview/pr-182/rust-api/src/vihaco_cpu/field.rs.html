4//! Struct operands for `struct_new`, `get_field`, and `set_field`, and their lowering.
5//!
6//! Structs are written by name (`struct_new @Point`, `get_field @Point.y`) or
7//! by struct-table index (`struct_new 0`, `get_field 0, 1`). Both forms are
8//! resolved against the section's struct table through [`StructLookup`] when
9//! the program is lowered, so the [`StructLayout`] and [`FieldSlot`] stored in
10//! the runtime instruction always agree with the table. The CPU then checks
11//! each heap object against them at runtime.
12
13use crate::{FieldSlot, StructLayout, cpu};
14use vihaco_abi::{Lower, Syntax};
15use vihaco_module::loader::MetadataLoweringContext;
16use vihaco_parser::Ident;
17
18/// A struct, by struct-table index or `@Struct` name.
19#[derive(vihaco_parser_derive::Parse, Clone, Debug, PartialEq)]
20#[syntax_class(value)]
21pub enum StructOperand {
22    #[pattern = "$0"]
23    Resolved(u32),
24    #[pattern = "`@` $0"]
25    Symbol(Ident),
26}
27
28/// A struct field, as `struct_index, slot` or `@Struct.field`.
29#[derive(vihaco_parser_derive::Parse, Clone, Debug, PartialEq)]
30#[syntax_class(value)]
31pub enum FieldOperand {
32    #[pattern = "$0 `,` $1"]
33    Resolved(u32, u32),
34    #[pattern = "`@` $0"]
35    Symbol(Ident),
36}
37
38/// Resolve structs and fields against the section's struct table.
39pub trait StructLookup {
40    /// The struct-table index of the struct named `name`.
41    fn struct_index(&self, name: &str) -> eyre::Result<u32>;
42    /// The number of fields of struct-table entry `struct_index`.
43    fn struct_field_count(&self, struct_index: u32) -> eyre::Result<u32>;
44    /// The slot index of `field` within struct-table entry `struct_index`.
45    fn field_index(&self, struct_index: u32, field: &str) -> eyre::Result<u32>;
46}
47
48impl StructLookup for MetadataLoweringContext {
49    fn struct_index(&self, name: &str) -> eyre::Result<u32> {
50        MetadataLoweringContext::struct_index(self, name)
51    }
52    fn struct_field_count(&self, struct_index: u32) -> eyre::Result<u32> {
53        MetadataLoweringContext::struct_field_count(self, struct_index)
54    }
55    fn field_index(&self, struct_index: u32, field: &str) -> eyre::Result<u32> {
56        MetadataLoweringContext::field_index(self, struct_index, field)
57    }
58}
59
60fn struct_layout<C: StructLookup + ?Sized>(
61    operand: StructOperand,
62    context: &C,
63) -> eyre::Result<StructLayout> {
64    let struct_id = match operand {
65        StructOperand::Resolved(index) => index,
66        StructOperand::Symbol(name) => context.struct_index(name.as_str())?,
67    };
68    Ok(StructLayout {
69        struct_id,
70        field_count: context.struct_field_count(struct_id)?,
71    })
72}
73
74fn field_slot<C: StructLookup + ?Sized>(
75    operand: FieldOperand,
76    context: &C,
77) -> eyre::Result<FieldSlot> {
78    match operand {
79        FieldOperand::Resolved(struct_id, slot) => {
80            let count = context.struct_field_count(struct_id)?;
81            eyre::ensure!(
82                slot < count,
83                "struct {struct_id} has no field slot {slot} ({count} fields)"
84            );
85            Ok(FieldSlot { struct_id, slot })
86        }
87        FieldOperand::Symbol(symbol) => {
88            let symbol = symbol.as_str();
89            let (struct_name, field) = symbol
90                .rsplit_once('.')
91                .filter(|(struct_name, field)| !struct_name.is_empty() && !field.is_empty())
92                .ok_or_else(|| {
93                    eyre::eyre!("field operand `@{symbol}` must have the form `@Struct.field`")
94                })?;
95            let struct_id = context.struct_index(struct_name)?;
96            Ok(FieldSlot {
97                struct_id,
98                slot: context.field_index(struct_id, field)?,
99            })
100        }
101    }
102}
103
104impl<C: StructLookup + ?Sized> Lower<C> for cpu::StructNew<Syntax> {
105    type Runtime = cpu::StructNew;
106    fn lower(self, context: &mut C) -> eyre::Result<Self::Runtime> {
107        Ok(cpu::StructNew(struct_layout(self.0, context)?))
108    }
109}
110
111impl<C: StructLookup + ?Sized> Lower<C> for cpu::GetField<Syntax> {
112    type Runtime = cpu::GetField;
113    fn lower(self, context: &mut C) -> eyre::Result<Self::Runtime> {
114        Ok(cpu::GetField(field_slot(self.0, context)?))
115    }
116}
117
118impl<C: StructLookup + ?Sized> Lower<C> for cpu::SetField<Syntax> {
119    type Runtime = cpu::SetField;
120    fn lower(self, context: &mut C) -> eyre::Result<Self::Runtime> {
121        Ok(cpu::SetField(field_slot(self.0, context)?))
122    }
123}

125#[cfg(test)]
126mod tests {
127    use super::*;
128    use vihaco_runtime::{Parse, chumsky::Parser as _};
129
130    /// Struct 0 is `Point { x, y }`; struct 1 is `Unit {}`.
131    struct Points;
132
133    impl StructLookup for Points {
134        fn struct_index(&self, name: &str) -> eyre::Result<u32> {
135            match name {
136                "Point" => Ok(0),
137                "Unit" => Ok(1),
138                _ => Err(eyre::eyre!("undefined struct `{name}`")),
139            }
140        }
141        fn struct_field_count(&self, struct_index: u32) -> eyre::Result<u32> {
142            match struct_index {
143                0 => Ok(2),
144                1 => Ok(0),
145                _ => Err(eyre::eyre!(
146                    "struct {struct_index} is outside the struct table"
147                )),
148            }
149        }
150        fn field_index(&self, struct_index: u32, field: &str) -> eyre::Result<u32> {
151            match (struct_index, field) {
152                (0, "x") => Ok(0),
153                (0, "y") => Ok(1),
154                _ => Err(eyre::eyre!("struct {struct_index} has no field `{field}`")),
155            }
156        }
157    }
158
159    fn lower<I>(text: &str) -> eyre::Result<I::Runtime>
160    where
161        I: for<'src> Parse<'src> + Lower<Points>,
162    {
163        I::parser()
164            .parse(text)
165            .into_result()
166            .map_err(|errors| eyre::eyre!("{errors:?}"))?
167            .lower(&mut Points)
168    }
169
170    fn slot(struct_id: u32, slot: u32) -> FieldSlot {
171        FieldSlot { struct_id, slot }
172    }
173
174    #[test]
175    fn struct_new_takes_its_field_count_from_the_table() {
176        let point = StructLayout {
177            struct_id: 0,
178            field_count: 2,
179        };
180        assert_eq!(
181            lower::<cpu::StructNew<Syntax>>("struct_new @Point").unwrap(),
182            cpu::StructNew(point)
183        );
184        assert_eq!(
185            lower::<cpu::StructNew<Syntax>>("struct_new 0").unwrap(),
186            cpu::StructNew(point)
187        );
188        assert!(lower::<cpu::StructNew<Syntax>>("struct_new @Line").is_err());
189        assert!(lower::<cpu::StructNew<Syntax>>("struct_new 7").is_err());
190    }
191
192    #[test]
193    fn field_operands_resolve_names_and_check_numeric_slots() {
194        assert_eq!(
195            lower::<cpu::GetField<Syntax>>("get_field @Point.y").unwrap(),
196            cpu::GetField(slot(0, 1))
197        );
198        assert_eq!(
199            lower::<cpu::GetField<Syntax>>("get_field 0, 1").unwrap(),
200            cpu::GetField(slot(0, 1))
201        );
202        assert_eq!(
203            lower::<cpu::SetField<Syntax>>("set_field @Point.x").unwrap(),
204            cpu::SetField(slot(0, 0))
205        );
206        let err = lower::<cpu::GetField<Syntax>>("get_field 0, 2").unwrap_err();
207        assert!(err.to_string().contains("no field slot 2"), "{err}");
208        assert!(lower::<cpu::GetField<Syntax>>("get_field 1, 0").is_err());
209        assert!(lower::<cpu::GetField<Syntax>>("get_field 9, 0").is_err());
210        assert!(lower::<cpu::GetField<Syntax>>("get_field @Point.z").is_err());
211        assert!(lower::<cpu::GetField<Syntax>>("get_field @Line.x").is_err());
212    }
213
214    #[test]
215    fn malformed_field_symbols_are_rejected() {
216        for symbol in ["@Point", "@.y", "@Point."] {
217            let err = lower::<cpu::GetField<Syntax>>(&format!("get_field {symbol}")).unwrap_err();
218            assert!(err.to_string().contains("@Struct.field"), "{err}");
219        }
220    }
