4//! Experimental host interface for programs that can park within an instruction.
5//!
6//! A driver resolves machine IDs to invocations. Checked drivers use the
7//! machine's `HasProgram` implementation to reach its execution state and PC.
8
9use vihaco_abi::word::Word;
10
11/// Authorization to fetch instructions for one invocation.
12#[derive(Debug, Clone, Copy, PartialEq, Eq)]
13pub enum RunCommand {
14    /// Run until a breakpoint, suspension, or halt.
15    Continue,
16    /// Retire one instruction, even if it takes several deliveries.
17    SingleStep,
18}
19
20/// Why the driver returned control to the host.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum ProgramStop<W> {
23    /// The originating instruction still owns in-flight work.
24    Pending { pc: u32, work: W },
25    /// A single-step command retired this instruction.
26    Retired { pc: u32 },
27    /// A breakpoint stopped execution at this instruction.
28    Breakpoint { pc: u32 },
29    /// The invocation has halted.
30    Halted { pc: u32 },
31}
32
33/// Owns program issue and resumption for one or more machines.
34///
35/// A delivery advances an existing instruction under its saved command.
36/// Ordinary execution faults travel through `eyre::Result`.
37pub trait ProgramDriver {
38    /// The machine selected by a host command.
39    type MachineId: Copy;
40
41    /// Identifies pending work for the selected machine.
42    type WorkId: Copy;
43
44    /// A backend-specific readiness event or completion payload.
45    type Delivery;
46
47    /// Enter the selected machine's loaded program with the supplied arguments.
48    fn start(&mut self, machine: Self::MachineId, arguments: &[Word]) -> eyre::Result<()>;
49
50    /// Issue instructions under the supplied command until the program stops.
51    fn run(
52        &mut self,
53        machine: Self::MachineId,
54        command: RunCommand,
55    ) -> eyre::Result<ProgramStop<Self::WorkId>>;
56
57    /// Deliver work to a parked instruction and honor its saved command.
58    fn deliver(
59        &mut self,
60        machine: Self::MachineId,
61        work: Self::WorkId,
62        delivery: Self::Delivery,
63    ) -> eyre::Result<ProgramStop<Self::WorkId>>;
64}
