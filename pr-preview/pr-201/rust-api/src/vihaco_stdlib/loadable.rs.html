4//! An independently owned image paired with its mutable execution machine.
5
6use vihaco_abi::{traits::Reset, word::Word};
7use vihaco_module::module::LocalModule;
8use vihaco_runtime::{Composite, Dispatch, HasProgram};
9
10use crate::{ExecutionState, Image, execution_state};
11
12/// A program image and the composite that executes its borrowed instructions.
13///
14/// Construct the image explicitly before pairing it with a machine. Only
15/// [`Image`] is supported; the image remains unchanged throughout an invocation.
16/// Ordinary outer structs can own several loadables with different instruction
17/// sets. Section selection and image construction remain caller responsibilities.
18#[derive(Debug)]
19pub struct Loadable<I, M: Composite> {
20    image: I,
21    machine: M,
22}
23
24impl<Inst, V, Ty, Info, M: Composite> Loadable<Image<Inst, V, Ty, Info>, M> {
25    /// Inspect the paired image without changing an invocation's program data.
26    pub fn image(&self) -> &Image<Inst, V, Ty, Info> {
27        &self.image
28    }
29
30    /// Inspect the execution machine and its devices.
31    pub fn machine(&self) -> &M {
32        &self.machine
33    }
34
35    /// Configure the execution machine or its devices.
36    ///
37    /// Direct changes to execution state can affect whether an invocation can
38    /// resume. The caller is responsible for keeping those changes consistent
39    /// with the paired image.
40    pub fn machine_mut(&mut self) -> &mut M {
41        &mut self.machine
42    }
43
44    /// Borrow the image and mutably borrow the machine for custom runners.
45    ///
46    /// Within [`execution_state::run`], pass `&image.module` and the machine
47    /// to [`execution_state::execute_instruction`] to drive a step. A runner
48    /// with its own dispatch context can use
49    /// [`execution_state::execute_instruction_with_context`]. Keep any direct
50    /// changes to invocation state consistent with the paired image.
51    pub fn parts_mut(&mut self) -> (&Image<Inst, V, Ty, Info>, &mut M) {
52        (&self.image, &mut self.machine)
53    }
54
55    pub(crate) fn breakpoint_parts_mut(&mut self) -> (&mut Image<Inst, V, Ty, Info>, &mut M) {
56        (&mut self.image, &mut self.machine)
57    }
58
59    /// Consume the owner, returning its image and machine without resetting them.
60    pub fn into_parts(self) -> (Image<Inst, V, Ty, Info>, M) {
61        (self.image, self.machine)
62    }
63}
64
65impl<Inst, V, Ty, Info, M, S> Loadable<Image<Inst, V, Ty, Info>, M>
66where
67    M: Composite + HasProgram<Program = ExecutionState<S>>,
68{
69    /// Pair an image with a machine, resetting its previous invocation state.
70    ///
71    /// Device state is retained. No parsing, lowering, or image validation is
72    /// performed here; construction of a valid image is the caller's job.
73    ///
74    /// When reusing a machine with a different image, call
75    /// [`Debugger::forget_breakpoints`](crate::debugger::Debugger::forget_breakpoints)
76    /// on each debugger before using it with the new image. Saved instructions
77    /// belong to the discarded image; retaining them can overwrite new code
78    /// when a breakpoint is cleared. To keep the old image usable, restore its
79    /// instructions with [`Debugger::clear_all`](crate::debugger::Debugger::clear_all)
80    /// before separating it from the machine instead.
81    pub fn new(image: Image<Inst, V, Ty, Info>, mut machine: M) -> Self {
82        machine.program_mut().reset();
83        Self { image, machine }
84    }
85
86    /// Install an already constructed image and reset execution state.
87    ///
88    /// Device state is retained. Image construction happens before this call;
89    /// mutations made by the caller while constructing it are not rolled back.
90    ///
91    /// After replacement succeeds, call
92    /// [`Debugger::forget_breakpoints`](crate::debugger::Debugger::forget_breakpoints)
93    /// on each debugger before using it again. This discards instructions saved
94    /// from the old image, which must never be restored into the replacement.
95    /// Do not forget them before this call: if replacement is rejected, the old
96    /// image still needs those instructions to clear or resume its breakpoints.
97    /// Invocation-only [`Self::reset`] calls retain the image and its breakpoints.
98    /// See the `debugger` Cargo example for the replacement and reuse sequence.
99    ///
100    /// # Errors
101    ///
102    /// Rejects replacement while an invocation is active, including one that
103    /// could resume after an execution error. Call [`Self::reset`] to abandon
104    /// that invocation first. Rejection leaves the current image and execution
105    /// state unchanged.
106    pub fn replace_image(&mut self, image: Image<Inst, V, Ty, Info>) -> eyre::Result<()> {
107        eyre::ensure!(
108            self.machine.program().is_halted(),
109            "cannot replace an active program"
110        );
111        self.image = image;
112        self.reset();
113        Ok(())
114    }
115
116    /// Initialize an invocation without executing instructions.
117    pub fn start(&mut self, arguments: &[Word]) -> eyre::Result<()> {
118        self.machine
119            .program_mut()
120            .start(&self.image.module, arguments)
121    }
122
123    /// Abandon the invocation, retaining the image and all device state.
124    pub fn reset(&mut self) {
125        self.machine.program_mut().reset();
126    }
127}

129impl<Inst, V, Ty, Info, M, S> Loadable<Image<Inst, V, Ty, Info>, M>
130where
131    M: HasProgram<Program = ExecutionState<S>>
132        + Dispatch<LocalModule<Inst, V, Ty, Info>, Instruction = Inst>,
133{
134    /// Start at the program entrypoint with the supplied arguments.
135    ///
136    /// Instructions and metadata borrow the image while dispatch mutates the
137    /// separate machine. No instruction cloning is required.
138    ///
139    /// # Errors
140    ///
141    /// Returns the runner's entry or execution errors. Earlier machine mutations
142    /// are retained; reset or resume according to the invocation's state.
143    pub fn execute(&mut self, arguments: &[Word]) -> eyre::Result<()> {
144        execution_state::execute(&self.image.module, &mut self.machine, arguments)
145    }
146
147    /// Continue the existing invocation against the same paired image.
148    ///
149    /// # Errors
150    ///
151    /// Returns the runner's state, instruction-fetch, or dispatch errors without
152    /// rolling back machine mutations. A halted invocation remains halted.
153    /// A suspended invocation also remains suspended; use
154    /// [`Self::resume_with_breaks`] to continue from a debugger breakpoint.
155    pub fn resume(&mut self) -> eyre::Result<()> {
156        execution_state::resume(&self.image.module, &mut self.machine)
157    }
158}
