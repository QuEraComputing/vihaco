4//! Struct, named-type, and type operands for the struct, array, `wrap`, and
5//! `unwrap` instructions, and their lowering.
6//!
7//! Structs are written by name (`struct_new @Point`, `get_field @Point.y`) or
8//! by struct-table index (`struct_new 0`, `get_field 0, 1`). Named types are
9//! written the same way (`wrap @location`, `wrap 0`). Array element types are
10//! ordinary types, where a struct or named type may also be named:
11//! `array_new struct @Point, 4`, `array_get type @location`. Every form is
12//! resolved against the section's struct table through [`StructLookup`] and
13//! its named-type table through [`NamedTypeLookup`] when the program is
14//! lowered, so the [`StructLayout`], [`FieldSlot`], named-type index, and
15//! element [`Type`] stored in the runtime instruction always agree with the
16//! tables. The CPU then checks each heap object against them at runtime.
17
18use crate::{FieldSlot, StructLayout, cpu};
19use chumsky::{error::Simple, extra, prelude::*};
20use vihaco_abi::{Lower, Syntax, Type};
21use vihaco_module::loader::MetadataLoweringContext;
22use vihaco_parser::{Ident, Parse};

24/// A type as written in source: any [`Type`], where struct and named types may
25/// also be named (`struct @Point`, `type @location`), at any nesting depth
26/// (`array array struct @Point`).
27#[derive(Clone, Debug, PartialEq)]
28pub enum TypeOperand {
29    /// A type with no names to resolve, such as `f64` or `array struct 0`.
30    Resolved(Type),
31    /// `array <elem>`, where the element type may contain names.
32    Array(Box<TypeOperand>),
33    /// `struct @Name`.
34    StructSymbol(Ident),
35    /// `type @name`.
36    NamedSymbol(Ident),
37}
38
39impl<'src> Parse<'src> for TypeOperand {
40    fn parser() -> impl Parser<'src, &'src str, Self, extra::Err<Simple<'src, char>>> {
41        // `recursive` needs cloneable parsers; `Parse::parser` is not `Clone`.
42        let resolved = Type::parser().boxed();
43        recursive(move |operand| {
44            let named_struct = just("struct")
45                .then(text::whitespace().at_least(1))
46                .then(just('@'))
47                .ignore_then(vihaco_parser::ident())
48                .map(|name| TypeOperand::StructSymbol(Ident(name)));
49            let named_type = just("type")
50                .then(text::whitespace().at_least(1))
51                .then(just('@'))
52                .ignore_then(vihaco_parser::ident())
53                .map(|name| TypeOperand::NamedSymbol(Ident(name)));
54            let array = just("array")
55                .then(text::whitespace().at_least(1))
56                .ignore_then(operand)
57                .map(|elem| match elem {
58                    // Keep fully resolved types in one piece.
59                    TypeOperand::Resolved(elem) => TypeOperand::Resolved(Type::array(elem)),
60                    elem => TypeOperand::Array(Box::new(elem)),
61                });
62            choice((
63                named_struct,
64                named_type,
65                array,
66                resolved.clone().map(TypeOperand::Resolved),
67            ))
68        })
69    }
70}

92/// A struct field, as `struct_index, slot` or `@Struct.field`.
93#[derive(vihaco_parser_derive::Parse, Clone, Debug, PartialEq)]
94#[syntax_class(value)]
95pub enum FieldOperand {
96    #[pattern = "$0 `,` $1"]
97    Resolved(u32, u32),
98    #[pattern = "`@` $0"]
99    Symbol(Ident),

102/// Resolve structs and fields against the section's struct table.
103pub trait StructLookup {
104    /// The struct-table index of the struct named `name`.
105    fn struct_index(&self, name: &str) -> eyre::Result<u32>;
106    /// The number of fields of struct-table entry `struct_index`.
107    fn struct_field_count(&self, struct_index: u32) -> eyre::Result<u32>;
108    /// The slot index of `field` within struct-table entry `struct_index`.
109    fn field_index(&self, struct_index: u32, field: &str) -> eyre::Result<u32>;
110}
111
112impl StructLookup for MetadataLoweringContext {
113    fn struct_index(&self, name: &str) -> eyre::Result<u32> {
114        MetadataLoweringContext::struct_index(self, name)
115    }
116    fn struct_field_count(&self, struct_index: u32) -> eyre::Result<u32> {
117        MetadataLoweringContext::struct_field_count(self, struct_index)
118    }
119    fn field_index(&self, struct_index: u32, field: &str) -> eyre::Result<u32> {
120        MetadataLoweringContext::field_index(self, struct_index, field)
121    }
122}

124/// Resolve named types against the section's named-type table.
125pub trait NamedTypeLookup {
126    /// The named-type-table index of the named type called `name`.
127    fn named_type_index(&self, name: &str) -> eyre::Result<u32>;
128    /// Check that `index` is an entry of the named-type table.
129    fn check_named_type(&self, index: u32) -> eyre::Result<()>;
130}
131
132impl NamedTypeLookup for MetadataLoweringContext {
133    fn named_type_index(&self, name: &str) -> eyre::Result<u32> {
134        MetadataLoweringContext::named_type_index(self, name)
135    }
136    fn check_named_type(&self, index: u32) -> eyre::Result<()> {
137        MetadataLoweringContext::check_named_type(self, index)
138    }
139}
140
141fn named_type<C: NamedTypeLookup + ?Sized>(
142    operand: NamedTypeOperand,
143    context: &C,
144) -> eyre::Result<u32> {
145    match operand {
146        NamedTypeOperand::Resolved(index) => {
147            context.check_named_type(index)?;
148            Ok(index)
149        }
150        NamedTypeOperand::Symbol(name) => context.named_type_index(name.as_str()),
151    }
152}

168fn struct_layout<C: StructLookup + ?Sized>(
169    operand: StructOperand,
170    context: &C,
171) -> eyre::Result<StructLayout> {
172    let struct_id = match operand {
173        StructOperand::Resolved(index) => index,
174        StructOperand::Symbol(name) => context.struct_index(name.as_str())?,
175    };
176    Ok(StructLayout {
177        struct_id,
178        field_count: context.struct_field_count(struct_id)?,
179    })
180}
181
182fn field_slot<C: StructLookup + ?Sized>(
183    operand: FieldOperand,
184    context: &C,
185) -> eyre::Result<FieldSlot> {
186    match operand {
187        FieldOperand::Resolved(struct_id, slot) => {
188            let count = context.struct_field_count(struct_id)?;
189            eyre::ensure!(
190                slot < count,
191                "struct {struct_id} has no field slot {slot} ({count} fields)"
192            );
193            Ok(FieldSlot { struct_id, slot })
194        }
195        FieldOperand::Symbol(symbol) => {
196            let symbol = symbol.as_str();
197            let (struct_name, field) = symbol
198                .rsplit_once('.')
199                .filter(|(struct_name, field)| !struct_name.is_empty() && !field.is_empty())
200                .ok_or_else(|| {
201                    eyre::eyre!("field operand `@{symbol}` must have the form `@Struct.field`")
202                })?;
203            let struct_id = context.struct_index(struct_name)?;
204            Ok(FieldSlot {
205                struct_id,
206                slot: context.field_index(struct_id, field)?,
207            })
208        }
209    }
210}

233/// Resolve struct and named-type names in a type, and check that every struct
234/// and named type it refers to (named or numeric) exists in its table.
235fn resolve_type<C: StructLookup + NamedTypeLookup + ?Sized>(
236    operand: TypeOperand,
237    context: &C,
238) -> eyre::Result<Type> {
239    let ty = match operand {
240        TypeOperand::Resolved(ty) => ty,
241        TypeOperand::Array(elem) => Type::array(resolve_type(*elem, context)?),
242        TypeOperand::StructSymbol(name) => Type::Struct(context.struct_index(name.as_str())?),
243        TypeOperand::NamedSymbol(name) => Type::Named(context.named_type_index(name.as_str())?),
244    };
245    let mut result = Ok(());
246    ty.visit_struct_refs(&mut |index| {
247        if result.is_ok() {
248            result = context.struct_field_count(index).map(drop);
249        }
250    });
251    ty.visit_named_refs(&mut |index| {
252        if result.is_ok() {
253            result = context.check_named_type(index);
254        }
255    });
256    result.map(|()| ty)
257}

259impl<C: StructLookup + NamedTypeLookup + ?Sized> Lower<C> for cpu::ArrayNew<Syntax> {
260    type Runtime = cpu::ArrayNew;
261    fn lower(self, context: &mut C) -> eyre::Result<Self::Runtime> {
262        Ok(cpu::ArrayNew(resolve_type(self.0, context)?, self.1))
263    }
264}

266macro_rules! lower_element_type {
267    ($($instruction:ident),* $(,)?) => {$(
268        impl<C: StructLookup + NamedTypeLookup + ?Sized> Lower<C> for cpu::$instruction<Syntax> {
269            type Runtime = cpu::$instruction;
270            fn lower(self, context: &mut C) -> eyre::Result<Self::Runtime> {
271                Ok(cpu::$instruction(resolve_type(self.0, context)?))
272            }
273        }
274    )*};
275}

284    /// Struct 0 is `Point { x, y }`; struct 1 is `Unit {}`. Named type 0 is
285    /// `location`.
286    struct Points;
287
288    impl NamedTypeLookup for Points {
289        fn named_type_index(&self, name: &str) -> eyre::Result<u32> {
290            match name {
291                "location" => Ok(0),
292                _ => Err(eyre::eyre!("undefined named type `{name}`")),
293            }
294        }
295        fn check_named_type(&self, index: u32) -> eyre::Result<()> {
296            eyre::ensure!(index == 0, "named type {index} is outside the table");
297            Ok(())
298        }
299    }
300
301    impl StructLookup for Points {
302        fn struct_index(&self, name: &str) -> eyre::Result<u32> {
303            match name {
304                "Point" => Ok(0),
305                "Unit" => Ok(1),
306                _ => Err(eyre::eyre!("undefined struct `{name}`")),
307            }
308        }
309        fn struct_field_count(&self, struct_index: u32) -> eyre::Result<u32> {
310            match struct_index {
311                0 => Ok(2),
312                1 => Ok(0),
313                _ => Err(eyre::eyre!(
314                    "struct {struct_index} is outside the struct table"
315                )),
316            }
317        }
318        fn field_index(&self, struct_index: u32, field: &str) -> eyre::Result<u32> {
319            match (struct_index, field) {
320                (0, "x") => Ok(0),
321                (0, "y") => Ok(1),
322                _ => Err(eyre::eyre!("struct {struct_index} has no field `{field}`")),
323            }
324        }
325    }
326
327    fn lower<I>(text: &str) -> eyre::Result<I::Runtime>
328    where
329        I: for<'src> Parse<'src> + Lower<Points>,
330    {
331        I::parser()
332            .parse(text)
333            .into_result()
334            .map_err(|errors| eyre::eyre!("{errors:?}"))?
335            .lower(&mut Points)
336    }

338    fn slot(struct_id: u32, slot: u32) -> FieldSlot {
339        FieldSlot { struct_id, slot }
